```ts
// An address on the relay (transport spec §2.3): `https://<relay>/<owner>/
// <doc>.geml#<block>`. An embed's `src=` is one; the daemon mirrors the
// document it names under `~/.gemworks/im/<relay>/<owner>/<doc>`. The relay's
// host stays in the address, so an embed across relays needs no other format.
// Pure: text in, parts out.

export interface Address {
  /** The relay's host, with its port when it has one: `relay.example`, `relay.example:8443`. */
  relay: string;
  owner: string;
  /** The document's path under the owner, `.geml` included: `venues/g-7.geml`. */
  doc: string;
  /** The block's id, without its `#`. */
  block?: string;
}

const SEGMENT = "[^/\\s#?]+";
const ADDRESS = new RegExp(`^https://(${SEGMENT})/(${SEGMENT})/((?:${SEGMENT}/)*${SEGMENT}\\.geml)(?:#([^\\s#]+))?$`);

/** The parts of an address, or undefined when the text is not one: another scheme, no owner, a document that is not `.geml`, a `.` or `..` segment, two `#`. */
export function parseAddress(text: string): Address | undefined {
  const m = ADDRESS.exec(text.trim());
  if (!m) return undefined;
  const relay = m[1]!, owner = m[2]!, doc = m[3]!, block = m[4];
  if ([owner, ...doc.split("/")].some((s) => s === "." || s === "..")) return undefined;
  const a: Address = { relay, owner, doc };
  if (block !== undefined) a.block = block;
  return a;
}

export function formatAddress(a: Address): string {
  return `https://${a.relay}/${a.owner}/${a.doc}${a.block === undefined ? "" : `#${a.block}`}`;
}

/**
 * Where the mirror keeps the document, relative to `~/.gemworks/im/`, with
 * `/` separators: the relay's host (a `:port` becomes `_port`, since a
 * Windows file name cannot hold a colon), then the owner, then the document.
 */
export function mirrorRelPath(a: Address): string {
  return `${a.relay.replace(/:/g, "_")}/${a.owner}/${a.doc}`;
}

export function sameDocument(a: Address, b: Address): boolean {
  return a.relay === b.relay && a.owner === b.owner && a.doc === b.doc;
}
```

- [ ] **Step 4: Write `src/core/im/embed.ts`**

```ts
// Reference, never copy (rule 10, §5.1): everything exists in one place as an
// addressed GEML block, and wherever else it appears it is an embed. Two
// resolutions — live (follows the source) and pinned (`rev` or `hash` names
// one revision); by permission — a reader gets the content, anyone else a
// stub that says whose private range it is and nothing of the content; within
// a budget — the host expands what fits (16 KB by default) and leaves the
// rest as addresses for the agent's read tool. Pure: the caller brings the
// access function; nothing here reads a file or a socket.
import { parseAddress } from "./address.js";

export interface Embed {
  src: string;
  /** A journal revision, when the embed is pinned to one. */
  rev?: number;
  /** The block's hash, when the embed is pinned to it. */
  hash?: string;
}

export const isPinned = (e: Embed): boolean => e.rev !== undefined || e.hash !== undefined;

/** What the viewer sees in place of a block they may not read: whose private range, which session and turn, and the one action. */
export interface Stub { from: string; session?: string; turn?: string; note: "你看不到"; action: "申请看" }

export interface Resolved {
  embed: Embed;
  text?: string;
  stub?: Stub;
  /** Why there is neither text nor stub: 不是地址, 找不到, or a pinned block that changed. */
  fault?: string;
}

/** What the caller knows about one block for one viewer: its text and its own hash when the viewer may read it, a denial otherwise, nothing when it does not exist. */
export type Access = (e: Embed, viewer: string) =>
  | { text: string; hash: string; owner: string; session?: string; turn?: string }
  | { denied: true; owner: string; session?: string; turn?: string }
  | undefined;

export const DEFAULT_BUDGET = 16384;

export function resolveEmbed(e: Embed, viewer: string, access: Access): Resolved {
  if (parseAddress(e.src) === undefined) return { embed: e, fault: "不是地址" };
  const got = access(e, viewer);
  if (got === undefined) return { embed: e, fault: "找不到" };
  if ("denied" in got) {
    const stub: Stub = { from: got.owner, note: "你看不到", action: "申请看" };
    if (got.session !== undefined) stub.session = got.session;
    if (got.turn !== undefined) stub.turn = got.turn;
    return { embed: e, stub };
  }
  if (e.hash !== undefined && e.hash !== got.hash) return { embed: e, fault: "钉住的块已变：哈希不符" };
  return { embed: e, text: got.text };
}

const bytes = (s: string): number => new TextEncoder().encode(s).length;

/**
 * The embeds expanded in order until the budget (UTF-8 bytes of text) would
 * be exceeded; the first that does not fit and every one after it are left
 * as addresses. A stub or a fault costs nothing and is always expanded.
 */
export function expandEmbeds(parts: readonly Embed[], viewer: string, access: Access, budget = DEFAULT_BUDGET): { expanded: Resolved[]; left: Embed[] } {
  const expanded: Resolved[] = [];
  let used = 0;
  for (let i = 0; i < parts.length; i++) {
    const r = resolveEmbed(parts[i]!, viewer, access);
    const cost = r.text === undefined ? 0 : bytes(r.text);
    if (used + cost > budget) return { expanded, left: parts.slice(i) };
    used += cost;
    expanded.push(r);
  }
  return { expanded, left: [] };
}
```

```ts
// Every IM document is a GEML document of `=== data {#id format=json}`
// blocks, one canonical JSON line each, chained the way the journal's
// revisions are (transport spec §2.2): a block's `hash` is the sha256 of its
// canonical JSON without `hash`, `prev` is the hash of the block before it in
// the same document, the first block's `prev` is null. Canonical JSON is one
// line with sorted keys, so CRLF on disk cannot change a hash; prose inside a
// block is normalized before it is sealed. Pure: text in, blocks and faults
// out; the hosts read and write the files.
import { createHash } from "node:crypto";
import { parse } from "@geml/geml";
import { canonical } from "../snapshot.js";
import type { JsonValue } from "../schema.js";

export interface Chained { prev: string | null; hash: string }

/** The sha256 of the value's canonical JSON, `hash` left out. */
export function hashOf(value: Record<string, JsonValue>): string {
  const { hash: _hash, ...rest } = value;
  return "sha256:" + createHash("sha256").update(canonical(rest), "utf8").digest("hex");
}

/** The value as a block on the chain: `prev` set, `hash` computed over everything else. */
export function seal<T extends Record<string, JsonValue>>(value: T, prev: string | null): T & Chained {
  const withPrev = { ...value, prev };
  return { ...withPrev, hash: hashOf(withPrev) };
}

/** Prose as it is sealed: LF line ends, no trailing blanks on a line, no trailing newlines. */
export function normalizeText(text: string): string {
  return text.replace(/\r\n?/g, "\n").split("\n").map((l) => l.replace(/[ \t]+$/, "")).join("\n").replace(/\n+$/, "");
}

export function renderDataBlock(id: string, value: JsonValue): string {
  return `=== data {#${id} format=json}\n${canonical(value)}\n===\n\n`;
}

export interface DataBlock { id: string; value: Record<string, JsonValue> }

const isObject = (v: unknown): v is Record<string, JsonValue> => typeof v === "object" && v !== null && !Array.isArray(v);

/** The document's data blocks with ids, in order; a parse error or a value that is not an object is a diagnostic. */
export function readDataBlocks(source: string): { blocks: DataBlock[]; diagnostics: string[] } {
  const doc = parse(source);
  const diagnostics = doc.diagnostics.filter((d) => d.severity === "error").map((d) => `line ${d.line}: ${d.message}`);
  const blocks: DataBlock[] = [];
  for (const b of doc.children) {
    if (b.kind !== "block" || b.type !== "data" || b.id === undefined) continue;
    if (!isObject(b.value)) { diagnostics.push(`#${b.id}: not a JSON object`); continue; }
    blocks.push({ id: b.id, value: b.value });
  }
  return { blocks, diagnostics };
}

/** The chain's faults, in order: a block without a hash, a hash that does not match its block, a prev that does not name the block before it. */
export function verifyChain(blocks: readonly DataBlock[]): string[] {
  const out: string[] = [];
  let last: string | null = null;
  for (const b of blocks) {
    const v = b.value;
    const hash = v["hash"];
    if (typeof hash !== "string") { out.push(`#${b.id}: no hash`); last = null; continue; }
    if (hashOf(v) !== hash) out.push(`#${b.id}: hash does not match`);
    if ((v["prev"] ?? null) !== last) out.push(`#${b.id}: prev does not name the block before it`);
    last = hash;
  }
  return out;
}
```

```ts
// Who speaks (§4, rule 1, rule 6): a person, an agent account (always shown
// with its owner's badge), a guest, or the host itself (system messages).
// The host stamps the speaker; this module only spells it. A message's
// source is which of the four ways it came to be (rule 6): typed by the
// person, drafted by the agent and sent by the person, said by the agent
// within its scope, or the agent quoting its owner. Pure.

export type Speaker =
  | { kind: "person"; id: string }
  | { kind: "agent"; name: string; owner: string }
  | { kind: "guest"; id: string }
  | { kind: "host" };

/** Rule 6's four sources: 本人亲打, agent 起草本人发出, agent 在权限内自己说的, agent 转述主人的话. */
export type Source = "self" | "drafted" | "agent" | "quoted";

/** An id or a name as the relay spells it: letters, digits, `_`, `-`. */
export const NAME = /^[\p{L}\p{N}_-]+$/u;

export function parseSpeaker(text: string): Speaker | undefined {
  if (text === "host") return { kind: "host" };
  const m = /^(person|agent|guest):(.+)$/s.exec(text);
  if (!m) return undefined;
  const rest = m[2]!;
  if (m[1] === "agent") {
    const at = rest.indexOf("@");
    if (at <= 0 || at === rest.length - 1) return undefined;
    const name = rest.slice(0, at), owner = rest.slice(at + 1);
    if (!NAME.test(name) || !NAME.test(owner)) return undefined;
    return { kind: "agent", name, owner };
  }
  if (!NAME.test(rest)) return undefined;
  return m[1] === "person" ? { kind: "person", id: rest } : { kind: "guest", id: rest };
}

export function formatSpeaker(s: Speaker): string {
  switch (s.kind) {
    case "person": return `person:${s.id}`;
    case "agent": return `agent:${s.name}@${s.owner}`;
    case "guest": return `guest:${s.id}`;
    case "host": return "host";
  }
}

/** The person responsible: a person is themself, an agent is its owner; a guest and the host are nobody's. */
export function personBehind(s: Speaker): string | undefined {
  return s.kind === "person" ? s.id : s.kind === "agent" ? s.owner : undefined;
}

/** An agent's name as it is always drawn (rule 6): with its owner's badge. */
export const agentLabel = (agentName: string, ownerName: string): string => `${agentName} · 代 ${ownerName}`;

/** The name-line tag of a drafted message (rule 6): the agent that drafted it. */
export const draftedTag = (agentName: string): string => `${agentName}起草`;

/** The tag on an agent's quotation of its owner (rule 6). */
export const QUOTED_TAG = "本人原话";

/** Why a speaker cannot claim a source: a person (or guest) is self or drafted, an agent is agent or quoted; the host's messages have no source to check. */
export function sourceFaults(s: Speaker, source: Source): string[] {
  if (s.kind === "host") return [];
  if (s.kind === "agent") return source === "agent" || source === "quoted" ? [] : ["agent 的消息不能标为本人亲打"];
  return source === "self" || source === "drafted" ? [] : ["本人的消息不能标为 agent 发言"];
}
```

```ts
// The two meanings of `@` (§6.4): `@名字` as the first word calls the agent —
// it gets a prompt; a run of names at the start is a broadcast, each called
// once. `@名字` in the middle of a sentence only mentions: no delivery, a
// 「被提到」 on the agent's board. `@` a person, and `@所有人`, only ever
// notify. A role address (`@空闲的`, `@审稿人`, `@#42的负责人`) resolves to one
// agent by the venue's policy, and the resolution is written back on the
// message. Pure: the roster and the resolver come from the caller.

export type RosterKind = "person" | "agent" | "role" | "everyone";
export interface RosterEntry { name: string; kind: RosterKind }

/** One `@` on a message as it is stored: the name, whether it called (first word) or only mentioned, and the agent a role resolved to. */
export interface Mention { name: string; call: boolean; resolved?: string }

export interface Mentions {
  /** Agents and roles at the start of the text, in order, each once. */
  calls: string[];
  /** Agents and roles anywhere else. */
  mentions: string[];
  /** People and 所有人, wherever they stand. */
  notified: string[];
  /** Names the roster does not have. */
  unknown: string[];
}

// A name runs to whitespace, another `@`, or punctuation; `@` counts only at the start or after whitespace or punctuation (not inside an e-mail address).
const NAME_CHAR = "[^\\s@,，。！？!?:：;；()（）]";
const TOKEN = new RegExp(`(?<![\\p{L}\\p{N}_.-])@(${NAME_CHAR}+)`, "gu");
const LEAD = new RegExp(`^(?:\\s*@${NAME_CHAR}+[\\s,，]*)+`, "u");

export function parseMentions(text: string, roster: readonly RosterEntry[]): Mentions {
  const kinds = new Map(roster.map((r) => [r.name, r.kind] as const));
  const lead = LEAD.exec(text)?.[0].length ?? 0;
  const out: Mentions = { calls: [], mentions: [], notified: [], unknown: [] };
  const once = (list: string[], name: string) => { if (!list.includes(name)) list.push(name); };
  for (const m of text.matchAll(TOKEN)) {
    const name = m[1]!, kind = kinds.get(name), atStart = (m.index ?? 0) < lead;
    if (name === "所有人" || kind === "everyone" || kind === "person") once(out.notified, name);
    else if (kind === "agent" || kind === "role") once(atStart ? out.calls : out.mentions, name);
    else once(out.unknown, name);
  }
  return out;
}

/** The agent a role address names right now, or undefined when nobody fits. */
export type RoleResolver = (role: string) => string | undefined;

/** The calls as mentions on the message: a role becomes the agent it resolved to (`resolved`); a role nobody fits is reported, not called. */
export function resolveCalls(calls: readonly string[], roster: readonly RosterEntry[], resolve: RoleResolver): { calls: Mention[]; unresolved: string[] } {
  const kinds = new Map(roster.map((r) => [r.name, r.kind] as const));
  const out: Mention[] = [], unresolved: string[] = [];
  for (const name of calls) {
    if (kinds.get(name) !== "role") { out.push({ name, call: true }); continue; }
    const to = resolve(name);
    if (to === undefined) unresolved.push(name);
    else out.push({ name, call: true, resolved: to });
  }
  return { calls: out, unresolved };
}

export const CALL_HINT = "在句首 = 叫它干活";
export const NOTIFY_HINT = "只是通知";

export interface Candidate { name: string; kind: RosterKind; presence?: string; hint: string }

/** The popover while typing `@` (§6.4): the roster entries whose name starts with the prefix, agents with their presence, roles with how they resolve. */
export function mentionCandidates(roster: readonly RosterEntry[], prefix: string, presenceOf: (name: string) => string | undefined, roleHint: (role: string) => string): Candidate[] {
  const out: Candidate[] = [];
  for (const r of roster) {
    if (!r.name.startsWith(prefix)) continue;
    const c: Candidate = { name: r.name, kind: r.kind, hint: r.kind === "agent" ? CALL_HINT : r.kind === "role" ? roleHint(r.name) : NOTIFY_HINT };
    const p = r.kind === "agent" ? presenceOf(r.name) : undefined;
    if (p !== undefined) c.presence = p;
    out.push(c);
  }
  return out;
}
```

```ts
// A message (§4, §6.3): one block in its venue's document, possibly carrying
// work. Its kind says what was said; delivery (delivery.ts) says how it
// reaches an agent — the two are separate (§6.1). The speaker, the time and
// the chain are the host's stamp (rule 1): whatever a draft claims about them
// is dropped. Attachments, evidence and the 「回到那一轮」 link are embeds,
// never copies (rule 10). Pure: drafts in, sealed messages out, text in and
// out as data blocks.
import type { JsonValue } from "../schema.js";
import { normalizeText, readDataBlocks, renderDataBlock, seal, type DataBlock } from "./chain.js";
import type { Embed } from "./embed.js";
import { formatSpeaker, parseSpeaker, sourceFaults, type Source, type Speaker } from "./identity.js";
import type { Mention } from "./mentions.js";

export type MessageKind = "chat" | "report" | "ask" | "signal" | "card" | "system";
export const KIND_WORDS: Readonly<Record<MessageKind, string>> = { chat: "闲聊", report: "汇报", ask: "询问", signal: "信号", card: "卡片", system: "系统" };

/** The option every ask ends in (§6.3): the person writes their own answer. */
export const WRITE_IN = "自己写";

/** An ask (§6.3): the question, its options (buttons), whom it asks, when it expires, what happens when nobody answers. */
export interface Ask { question: string; options: string[]; who: string; expires?: string; fallback?: string }

/** What a client submits; the host stamps the rest. */
export interface Draft {
  venue: string;
  kind: MessageKind;
  text: string;
  parts?: Embed[];
  /** A report's evidence (§6.3): embeds of facts the host observed, drawn as 「已核实」. */
  evidence?: Embed[];
  mentions?: Mention[];
  source?: Source;
  hops?: number;
  /** A report's opinion, drawn apart from its evidence with the 「意见」 tag. */
  opinion?: string;
  /** 「回到那一轮」: the session turn's local address, resolved only on the owner's machine (rule 4). */
  turn?: string;
  ask?: Ask;
  /** A card message's card (§6.3): the task card's or deliverable's address; the card is the thing, the message a pointer. */
  card?: string;
  /** The thread's anchor, when the message is in one (§6.7). */
  thread?: string;
  replyTo?: string;
}

/** The host's stamp: the block's id, the speaker the token names, the time, the chain, and the agent's session when there is one. */
export interface Stamp { id: string; speaker: Speaker; at: string; prev: string | null; session?: string }

export interface Message {
  id: string;
  venue: string;
  kind: MessageKind;
  text: string;
  speaker: string;
  session?: string;
  at: string;
  mentions: Mention[];
  parts: Embed[];
  evidence: Embed[];
  source: Source;
  hops: number;
  opinion?: string;
  turn?: string;
  ask?: Ask;
  card?: string;
  thread?: string;
  replyTo?: string;
  prev: string | null;
  hash: string;
}

/** Who may send a kind (§6.3): a report, an ask and a signal only an agent; a system message only the host; chat and card anyone but the host. */
export function kindAllowed(kind: MessageKind, speaker: Speaker): boolean {
  switch (kind) {
    case "report": case "ask": case "signal": return speaker.kind === "agent";
    case "system": return speaker.kind === "host";
    case "chat": case "card": return speaker.kind !== "host";
  }
}

/** Why a draft cannot be stamped, or nothing. */
export function draftFaults(d: Draft, speaker: Speaker): string[] {
  const out: string[] = [];
  if (!kindAllowed(d.kind, speaker)) out.push(`${KIND_WORDS[d.kind]}：${formatSpeaker(speaker)} 不能发这种消息`);
  if (d.kind === "ask" && (d.ask === undefined || d.ask.question.trim() === "" || d.ask.who.trim() === "")) out.push("询问：缺问题或问谁");
  if (d.kind === "card" && (d.card === undefined || d.card === "")) out.push("卡片：缺卡的地址");
  if ((d.kind === "chat" || d.kind === "report") && d.text.trim() === "" && !(d.parts !== undefined && d.parts.length > 0)) out.push("正文为空");
  if (d.turn !== undefined && speaker.kind !== "agent") out.push("只有 agent 的消息能链接回那一轮");
  if (d.source !== undefined) out.push(...sourceFaults(speaker, d.source));
  return out;
}

const OPTIONAL = ["opinion", "turn", "card", "thread", "replyTo"] as const;

function askOf(a: Ask): Ask {
  const out: Ask = { question: a.question, options: a.options.includes(WRITE_IN) ? [...a.options] : [...a.options, WRITE_IN], who: a.who };
  if (a.expires !== undefined) out.expires = a.expires;
  if (a.fallback !== undefined) out.fallback = a.fallback;
  return out;
}

/**
 * The message as the host stamps it (rule 1): id, speaker, time, session and
 * chain from the stamp, whatever the draft says; the text normalized (LF, no
 * trailing blanks); the source defaulting to the speaker's kind; an ask's
 * options always ending in 自己写. Only the fields a message has are copied,
 * so a draft cannot smuggle a key in.
 */
export function stamp(d: Draft, s: Stamp): Message {
  const m: Omit<Message, "prev" | "hash"> = {
    id: s.id, venue: d.venue, kind: d.kind, text: normalizeText(d.text),
    speaker: formatSpeaker(s.speaker), at: s.at,
    mentions: d.mentions ?? [], parts: d.parts ?? [], evidence: d.evidence ?? [],
    source: d.source ?? (s.speaker.kind === "agent" ? "agent" : "self"), hops: d.hops ?? 0,
  };
  if (s.session !== undefined) m.session = s.session;
  for (const k of OPTIONAL) { const v = d[k]; if (v !== undefined) m[k] = v; }
  if (d.ask !== undefined) m.ask = askOf(d.ask);
  return seal(m as unknown as Record<string, JsonValue>, s.prev) as unknown as Message;
}

export function renderMessageBlock(m: Message): string {
  return renderDataBlock(m.id, m as unknown as JsonValue);
}

const isObject = (v: unknown): v is Record<string, JsonValue> => typeof v === "object" && v !== null && !Array.isArray(v);
const arrayOf = (v: JsonValue | undefined): JsonValue[] => (Array.isArray(v) ? v : []);

function messageOf(b: DataBlock): Message | string {
  const v = b.value;
  const str = (k: string): string | undefined => (typeof v[k] === "string" ? (v[k] as string) : undefined);
  const kind = str("kind");
  if (kind === undefined || !(kind in KIND_WORDS)) return "kind missing or unknown";
  if (str("id") !== b.id) return "id does not match the block";
  const speaker = str("speaker");
  if (speaker === undefined || parseSpeaker(speaker) === undefined) return "speaker missing or malformed";
  const venue = str("venue"), text = str("text"), at = str("at"), hash = str("hash");
  if (venue === undefined || text === undefined || at === undefined || hash === undefined) return "missing fields";
  const m: Message = {
    id: b.id, venue, kind: kind as MessageKind, text, speaker, at, hash,
    prev: typeof v["prev"] === "string" ? v["prev"] : null,
    mentions: arrayOf(v["mentions"]) as unknown as Mention[], parts: arrayOf(v["parts"]) as unknown as Embed[], evidence: arrayOf(v["evidence"]) as unknown as Embed[],
    source: (str("source") ?? "self") as Source, hops: typeof v["hops"] === "number" ? v["hops"] : 0,
  };
  const session = str("session");
  if (session !== undefined) m.session = session;
  for (const k of OPTIONAL) { const s = str(k); if (s !== undefined) m[k] = s; }
  const ask = v["ask"];
  if (ask !== undefined) {
    if (!isObject(ask) || typeof ask["question"] !== "string" || typeof ask["who"] !== "string" || !Array.isArray(ask["options"])) return "ask malformed";
    m.ask = ask as unknown as Ask;
  }
  return m;
}

/** The document's messages in order; a block that is no message is a diagnostic and skipped. The chain is verified separately (chain.ts). */
export function readMessages(source: string): { messages: Message[]; diagnostics: string[] } {
  const { blocks, diagnostics } = readDataBlocks(source);
  const messages: Message[] = [];
  for (const b of blocks) {
    const m = messageOf(b);
    if (typeof m === "string") diagnostics.push(`#${b.id}: ${m}`);
    else messages.push(m);
  }
  return { messages, diagnostics };
}
```

```ts
// Who may ask what of an agent (§7): a sender's level toward a target agent
// is the relation between the person behind the sender and the agent's owner
// (§7.1); the level picks a tool scope (§7.2) that only a recorded grant can
// widen (§7.3); an agent-to-agent chain of prompts stops after three hops and
// every venue caps how often agents speak (§7.4). The budgets are per turn,
// in credits (the platform's one unit, §15 item 16); the defaults stand until
// the owner sets theirs — §7.2 says only that a guest's is small. Pure.
import { personBehind, type Speaker } from "./identity.js";

export type Level = "owner" | "trusted" | "member" | "guest";
export const LEVEL_WORDS: Readonly<Record<Level, string>> = { owner: "主人", trusted: "信任", member: "成员", guest: "访客" };
/** Lower first: the queue's priority and the strictest reading of a scope. */
export const LEVEL_RANK: Readonly<Record<Level, number>> = { owner: 0, trusted: 1, member: 2, guest: 3 };

/** What a target agent's owner has said about people: who they are, whom they trust, who shares a venue with the agent. */
export interface Relations { owner: string; trusted: readonly string[]; members: readonly string[] }

/** The sender's level toward the target (§7.1): the person behind an agent is its owner; a guest and the host have nobody behind them. */
export function levelOf(sender: Speaker, r: Relations): Level {
  const p = personBehind(sender);
  if (p === undefined) return "guest";
  if (p === r.owner) return "owner";
  if (r.trusted.includes(p)) return "trusted";
  if (r.members.includes(p)) return "member";
  return "guest";
}

/** Where a run may happen: in the workflow as the owner's own calls do, in a sandbox, or not at all. */
export type Run = "workflow" | "sandbox" | "none";

/** What a level may do (§7.2). Reading and answering are always allowed. `budget` is credits per turn, null for the owner's own cap. */
export interface ToolScope { write: boolean; run: Run; budget: number | null }

/** The default per-turn budgets in credits, until the owner sets theirs. */
export const DEFAULT_BUDGETS: Readonly<Record<Exclude<Level, "owner">, number>> = { trusted: 100, member: 20, guest: 5 };

export const DEFAULT_SCOPES: Readonly<Record<Level, ToolScope>> = {
  owner: { write: true, run: "workflow", budget: null },
  trusted: { write: false, run: "sandbox", budget: DEFAULT_BUDGETS.trusted },
  member: { write: false, run: "none", budget: DEFAULT_BUDGETS.member },
  guest: { write: false, run: "none", budget: DEFAULT_BUDGETS.guest },
};

/** A recorded grant (§7.3): to one person or everyone in one venue, what it widens, since when, until when (none: for good). Every grant is human evidence. */
export interface Grant { to: { person?: string; venue?: string }; widen: Partial<ToolScope>; at: string; until?: string }

const RUN_RANK: Readonly<Record<Run, number>> = { none: 0, sandbox: 1, workflow: 2 };
const widerRun = (a: Run, b: Run): Run => (RUN_RANK[a] >= RUN_RANK[b] ? a : b);
const widerBudget = (a: number | null, b: number | null): number | null => (a === null || b === null ? null : Math.max(a, b));

/** When a grant 「允许此人」 runs out: seven days by default (§7.3), as an ISO time without milliseconds. */
export function grantUntil(at: string, days = 7): string {
  return new Date(Date.parse(at) + days * 86_400_000).toISOString().replace(/\.\d{3}Z$/, "Z");
}

/** The level's scope, widened by every grant that names this person or this venue and has not run out; a grant never narrows. */
export function scopeFor(level: Level, o: { person?: string; venue?: string; grants?: readonly Grant[]; budgets?: Partial<Record<Level, number>>; now: string }): ToolScope {
  const base = DEFAULT_SCOPES[level];
  const set = o.budgets?.[level];
  let scope: ToolScope = { ...base, budget: level === "owner" ? null : set ?? base.budget };
  for (const g of o.grants ?? []) {
    const named = (g.to.person !== undefined && g.to.person === o.person) || (g.to.venue !== undefined && g.to.venue === o.venue);
    if (!named || (g.until !== undefined && g.until <= o.now)) continue;
    scope = {
      write: scope.write || g.widen.write === true,
      run: g.widen.run === undefined ? scope.run : widerRun(scope.run, g.widen.run),
      budget: g.widen.budget === undefined ? scope.budget : widerBudget(scope.budget, g.widen.budget),
    };
  }
  return scope;
}

/** What a tool does, as the host classes it: reads, answers (speaks), runs a command, writes a file. */
export type ToolClass = "read" | "answer" | "run" | "write";
export type Decision = "allow" | "sandbox" | "refuse";

/** What the scope says to a call of that class; `sandbox` means the host runs it in the guest room, not the workspace (§7.5). */
export function decide(scope: ToolScope, cls: ToolClass): Decision {
  switch (cls) {
    case "read": case "answer": return "allow";
    case "write": return scope.write ? "allow" : "refuse";
    case "run": return scope.run === "workflow" ? "allow" : scope.run === "sandbox" ? "sandbox" : "refuse";
  }
}

/** The scope as the envelope writes it (§6.5). */
export function scopeWords(scope: ToolScope): string {
  if (scope.write && scope.run === "workflow") return "工作流允许的一切";
  if (scope.run !== "none") return "读、在沙箱里跑";
  return "只读";
}

export const HOP_LIMIT = 3;

/** The hops of a new prompt (§6.5): none when a person raised it; one more than the prompt the raising agent is handling. */
export const hopsOf = (handling: number | undefined): number => (handling === undefined ? 0 : handling + 1);

export const overHops = (hops: number, limit = HOP_LIMIT): boolean => hops > limit;

/** The system message when a prompt degrades to a plain message (§6.5). */
export const hopsSystemText = (agent: string, hops: number, limit: number): string =>
  `给 ${agent} 的指令是第 ${hops} 跳，超过上限 ${limit} 跳，按普通消息投递，没有进它的队列`;

/** A venue's caps on agents' messages per minute (§7.4): all agents together, and each agent. */
export interface Density { venuePerMinute: number; agentPerMinute: number }
export const DEFAULT_DENSITY: Density = { venuePerMinute: 30, agentPerMinute: 6 };

const MINUTE = 60_000;

/** Whether an agent may speak now, given the venue's recent messages. People's messages and anything older than a minute do not count. */
export function densityAllows(recent: readonly { speaker: string; at: string }[], agent: string, now: string, d: Density): { ok: true } | { ok: false; reason: string } {
```

```ts
// Where messages happen (§6.2): a DM (the session itself), a guest DM (a
// guest and an agent, the owner a silent member), a group (people and agents,
// its own policy, an announce, a decision log, a bot density), a thread
// (anchored on a message or a thing, inheriting the group it shows in, or
// independent and the anchor owner's), a subscription channel (signals only).
// The announce and the decisions reach an agent as live embeds (rule 3).
// Pure: records in, records and words out; nothing here knows a file.
import type { Embed } from "./embed.js";
import { DEFAULT_DENSITY, type Density } from "./sender.js";

export type VenueKind = "dm" | "guest-dm" | "group" | "thread" | "channel";
export const VENUE_WORDS: Readonly<Record<VenueKind, string>> = { dm: "私信", "guest-dm": "访客私信", group: "群", thread: "线程", channel: "订阅频道" };

/** Whose `@` becomes a prompt for an agent here (rule 2): anyone's, only the listed people's and agents', or no other agent's. */
export type PromptFrom = "anyone" | "allowlist" | "nobody";

export interface Policy {
  promptFrom: PromptFrom;
  /** Speakers (`person:…`, `agent:…`) whose prompts are taken when `promptFrom` is `allowlist`. */
  allow: string[];
  density: Density;
  /** 「回群前先给我看」: an agent's reply becomes a draft card for its owner first (§6.5). */
  draftFirst: boolean;
  hopLimit: number;
}

export const DEFAULT_POLICY: Policy = { promptFrom: "anyone", allow: [], density: DEFAULT_DENSITY, draftFirst: false, hopLimit: 3 };

/** The group's pinned text (§6.8): versioned; every change is a system message. */
export interface Announce { text: string; version: number; at: string; by: string }

export interface Venue {
  id: string;
  kind: VenueKind;
  name: string;
  /** The venue document's address on the relay, `https://<relay>/<owner>/venues/<id>.geml`. */
  address: string;
  /** Members: people by id, agents by their shown name. */
  people: string[];
  agents: string[];
  policy: Policy;
  announce?: Announce;
  /** The decision log (§6.8): addresses of the messages marked 决策, oldest first. */
  decisions: string[];
  /** A thread's anchor (§6.7) and the group it shows in; an independent thread has no parent. */
  anchor?: string;
  parent?: string;
  /** A guest DM's agent, by its shown name. */
  host?: string;
}

export const ANNOUNCE_LIMIT = 2000;

/** The announce replaced (§6.8): one more version and the system message that says so; too long is a fault. The input is not changed. */
export function setAnnounce(v: Venue, text: string, by: string, at: string): { venue: Venue; system: string } | { fault: string } {
  const length = [...text].length;
  if (length > ANNOUNCE_LIMIT) return { fault: `群公告最多两千字（现在 ${length} 字）` };
  const version = (v.announce?.version ?? 0) + 1;
  return { venue: { ...v, announce: { text, version, at, by } }, system: `${by} 改了群公告（第 ${version} 版）` };
}

/** A message marked 决策 (§6.8): appended to the log once. */
export function markDecision(v: Venue, messageAddress: string): Venue {
  return v.decisions.includes(messageAddress) ? v : { ...v, decisions: [...v.decisions, messageAddress] };
}

/** What the host injects when an agent handles a prompt from this venue (rule 3, §6.8): the announce and the last few decisions, as live embeds. */
export function contextSegment(v: Venue, recent = 5): Embed[] {
  const out: Embed[] = [];
  if (v.announce !== undefined) out.push({ src: `${v.address}#announce` });
  for (const d of v.decisions.slice(-recent)) out.push({ src: d });
  return out;
}

/** A thread (§6.7): shown in a group, it is that group's; shown nowhere, it is the anchor owner's own venue under the default policy. */
export function threadVenue(o: { id: string; anchor: string; title: string; address: string; shownIn?: Venue; owner: string }): Venue {
  const base = { id: o.id, kind: "thread" as const, name: o.title, address: o.address, decisions: [], anchor: o.anchor };
  if (o.shownIn !== undefined) return { ...base, people: [...o.shownIn.people], agents: [...o.shownIn.agents], policy: o.shownIn.policy, parent: o.shownIn.id };
  return { ...base, people: [o.owner], agents: [], policy: DEFAULT_POLICY };
}

/** What the 「消息」 page knows about a venue right now, beyond the venue itself. */
export interface RowState {
  unread: number;
  /** A thread: which agent works in it and the card's state. */
  working?: { agent: string; card?: string };
  /** A channel: its stage and today's count. */
  stage?: string;
  today?: number;
}

/** One row of the venue list (§6.2): kind, name, one line of state, unread. */
export function listRow(v: Venue, s: RowState): { kind: string; name: string; status: string; unread: number } {
  let status: string;
  switch (v.kind) {
    case "group": status = `${v.people.length} 人 · ${v.agents.length} 个 agent`; break;
    case "guest-dm": status = `${v.host ?? "没有 agent"} 接待`; break;
    case "thread": status = s.working === undefined ? "没人在干" : `${s.working.agent} 在干${s.working.card === undefined ? "" : ` · ${s.working.card}`}`; break;
    case "channel": status = `${s.stage ?? ""} · 今天 ${s.today ?? 0} 条`; break;
    case "dm": status = "就是会话本身"; break;
  }
  return { kind: VENUE_WORDS[v.kind], name: v.name, status, unread: s.unread };
}
```

```ts
// The envelope (§6.5, transport §3): the one line the host writes above a
// prompt it delivers, so what the model knows about the sender's identity,
// its permissions and where the reply goes comes from the host, never from
// the text. The same function serves the relay, the daemon and every host.
// Pure.
import { LEVEL_WORDS, scopeWords, type Level, type ToolScope } from "./sender.js";
import { VENUE_WORDS, type VenueKind } from "./venue.js";

export interface EnvelopeInput {
  venueKind: VenueKind;
  venueName: string;
  senderName: string;
  level: Level;
  scope: ToolScope;
  /** How many items wait behind this one. */
  queued: number;
  /** The venue or the agent has 「回群前先给我看」 on (§6.5). */
  draftFirst?: boolean;
}

/** The venue as the envelope names it: a group, thread or channel by name; a DM or guest DM by kind alone. */
export function venueLabel(kind: VenueKind, name: string): string {
  return kind === "dm" || kind === "guest-dm" ? VENUE_WORDS[kind] : `${VENUE_WORDS[kind]}「${name}」`;
}

/** Where the final answer goes (§6.5), and whether the owner sees it first. */
export function replyWords(kind: VenueKind, draftFirst: boolean): string {
  switch (kind) {
    case "group": return draftFirst ? "回复先给主人过目，再发回该群" : "回复将发回该群，成员可见";
    case "thread": return draftFirst ? "回复先给主人过目，再发回该线程" : "回复将发回该线程，成员可见";
    case "guest-dm": return draftFirst ? "回复先给主人过目，再发给访客" : "回复将发给访客";
    case "dm": return "回复在私信里";
    case "channel": return "回复不会发出";
  }
}

export function envelopeLine(e: EnvelopeInput): string {
  const parts = [
    `来自 ${venueLabel(e.venueKind, e.venueName)}`,
    `${e.senderName}（${LEVEL_WORDS[e.level]}）`,
    `权限：${scopeWords(e.scope)}`,
    replyWords(e.venueKind, e.draftFirst === true),
    `队列中还有 ${e.queued} 条`,
  ];
  return `[${parts.join(" · ")}]`;
}

/** The prompt as the model sees it: the envelope, a line break, the text. */
export const withEnvelope = (line: string, text: string): string => `${line}\n${text}`;
```

```ts
// One queue per agent account (§6.5), kept by the relay, outliving any
// session: each item a prompt with its sender's level, its venue, its time,
// its expiry and its hops. Priority is 主人 > 信任 > 成员 > 访客, and every
// person before any agent; within a level first come first served. Items are
// delivered only between turns; an item read in and one answered are later
// marker blocks, outside the chain (the relay appends items, the daemon
// marks). An item past its expiry is dropped with a system message to its
// sender. Pure: records in, records and words out, text in and out as data
// blocks.
import type { JsonValue } from "../schema.js";
import { readDataBlocks, renderDataBlock, seal } from "./chain.js";
import type { Embed } from "./embed.js";
import { LEVEL_RANK, type Level } from "./sender.js";
import type { VenueKind } from "./venue.js";

export interface QueueItem {
  /** The client's id for the item (transport §2.5: idempotent appends). */
  id: string;
  /** Its place in the queue's chain, 1-based. */
  n: number;
  prompt: string;
  /** The sender as a speaker string, and as shown. */
  sender: string;
  senderName: string;
  level: Level;
  venue: string;
  venueKind: VenueKind;
  venueName: string;
  /** The message that raised it, by address. */
  message?: string;
  at: string;
  expires?: string;
  hops: number;
  /** A forwarded context pack (§6.6): embeds, resolved by the receiver's permission. */
  context?: Embed[];
  prev: string | null;
  hash: string;
}

export type ItemDraft = Omit<QueueItem, "n" | "prev" | "hash">;

/** The daemon's mark on an item: read in by the session, or answered (with the reply's address). */
export interface Marker { kind: "read" | "answered"; n: number; item: string; at: string; reply?: string }

export interface Queue { items: QueueItem[]; markers: Marker[] }
export const EMPTY_QUEUE: Queue = { items: [], markers: [] };

/** Priority (§6.5): lower first; a person's prompt by its level, an agent's behind every person's, whatever its owner's level. */
export function rankOf(item: Pick<QueueItem, "sender" | "level">): number {
  return item.sender.startsWith("agent:") ? 4 : LEVEL_RANK[item.level];
}

/** The object without its undefined fields, so canonical JSON never sees one. */
function defined<T extends object>(o: T): T {
  return Object.fromEntries(Object.entries(o).filter(([, v]) => v !== undefined)) as T;
}

export function enqueue(q: Queue, d: ItemDraft): { queue: Queue; item: QueueItem } {
  const last = q.items[q.items.length - 1];
  const item = seal(defined({ ...d, n: q.items.length + 1 }) as unknown as Record<string, JsonValue>, last === undefined ? null : last.hash) as unknown as QueueItem;
  return { queue: { items: [...q.items, item], markers: q.markers }, item };
}

const isExpired = (item: QueueItem, now: string): boolean => item.expires !== undefined && item.expires <= now;

export type ItemState = "queued" | "read" | "answered" | "expired";

export function stateOf(q: Queue, id: string, now: string): ItemState | undefined {
  const item = q.items.find((i) => i.id === id);
  if (item === undefined) return undefined;
  if (q.markers.some((m) => m.kind === "answered" && m.item === id)) return "answered";
  if (q.markers.some((m) => m.kind === "read" && m.item === id)) return "read";
  return isExpired(item, now) ? "expired" : "queued";
}

/** The items still waiting, in delivery order: by rank, then by arrival. */
export function pending(q: Queue, now: string): QueueItem[] {
  return q.items.filter((i) => stateOf(q, i.id, now) === "queued").sort((a, b) => rankOf(a) - rankOf(b) || a.n - b.n);
}

/** The item to deliver next, and the unmarked items that ran out meanwhile (each owed a system message, `expiredSystemText`). */
export function next(q: Queue, now: string): { item?: QueueItem; dropped: QueueItem[] } {
  const dropped = q.items.filter((i) => stateOf(q, i.id, now) === "expired");
  const item = pending(q, now)[0];
  return item === undefined ? { dropped } : { item, dropped };
}

/** The item's 1-based place among the waiting, or undefined when it is not waiting. */
export function positionOf(q: Queue, id: string, now: string): number | undefined {
  const i = pending(q, now).findIndex((item) => item.id === id);
  return i === -1 ? undefined : i + 1;
}

function mark(q: Queue, m: Omit<Marker, "n">): Queue {
  return { items: q.items, markers: [...q.markers, defined({ ...m, n: q.markers.length + 1 })] };
}

export const markRead = (q: Queue, id: string, at: string): Queue => mark(q, { kind: "read", item: id, at });

export function markAnswered(q: Queue, id: string, at: string, reply?: string): Queue {
  return mark(q, reply === undefined ? { kind: "answered", item: id, at } : { kind: "answered", item: id, at, reply });
}

/** The system message a sender gets when their prompt ran out undelivered (§6.5). */
export const expiredSystemText = (item: QueueItem): string => `过期未投递：${item.venueName} 里 ${item.at} 给它的那条过了 ${item.expires ?? ""}，已丢弃`;

/** The line under a message that only mentioned an agent (§6.5). */
export const MENTION_STATUS = "提及 · 没投递";

/** What the agent is up to, for the sender's line: whether any session of it is alive, and what it is doing (presence words). */
export interface AgentNow { alive: boolean; doing?: string }

/** The sender's status line under their message (§6.5): queued with the place and what the agent does, read in, answered, offline with the expiry, or expired. */
export function senderStatus(q: Queue, id: string, now: string, agent: AgentNow): string | undefined {
  const state = stateOf(q, id, now);
  const item = q.items.find((i) => i.id === id);
  if (state === undefined || item === undefined) return undefined;
  switch (state) {
    case "answered": return "已回应";
    case "read": return `已读入 ✓✓（${q.markers.find((m) => m.kind === "read" && m.item === id)?.at ?? ""}）`;
    case "expired": return "过期，没投递";
    case "queued": {
      if (!agent.alive) return `离线，已留言${item.expires === undefined ? "" : ` · 过期 ${item.expires}`}`;
      const ahead = (positionOf(q, id, now) ?? 1) - 1;
      return `点名 · 排队中（前面 ${ahead} 条${agent.doing === undefined ? "" : ` · ${agent.doing}`}）`;
    }
  }
}

/** The queue as its document: the items in chain order, then the markers (`read-N`, `answered-N`). */
export function renderQueue(q: Queue): string {
  let out = "";
  for (const i of q.items) out += renderDataBlock(i.id, i as unknown as JsonValue);
  for (const m of q.markers) out += renderDataBlock(`${m.kind}-${m.n}`, { kind: m.kind, item: m.item, at: m.at, ...(m.reply === undefined ? {} : { reply: m.reply }) });
  return out;
}

/** The queue read back: items by `n`, markers by their block number; a block that is neither is a diagnostic. */
export function readQueue(source: string): { queue: Queue; diagnostics: string[] } {
  const { blocks, diagnostics } = readDataBlocks(source);
  const items: QueueItem[] = [], markers: Marker[] = [];
  for (const b of blocks) {
    const v = b.value;
    const marker = /^(read|answered)-([1-9]\d*)$/.exec(b.id);
    if (marker && (v["kind"] === "read" || v["kind"] === "answered") && typeof v["item"] === "string" && typeof v["at"] === "string") {
      const m: Marker = { kind: v["kind"], n: Number(marker[2]), item: v["item"], at: v["at"] };
      if (typeof v["reply"] === "string") m.reply = v["reply"];
      markers.push(m);
    } else if (typeof v["prompt"] === "string" && typeof v["n"] === "number" && v["id"] === b.id && typeof v["hash"] === "string") {
      items.push(v as unknown as QueueItem);
    } else diagnostics.push(`#${b.id}: neither an item nor a marker`);
  }
  items.sort((a, b) => a.n - b.n);
  markers.sort((a, b) => a.n - b.n);
  return { queue: { items, markers }, diagnostics };
}
```

```ts
// How a stamped message reaches agents (§6.1: the kind says what was said,
// delivery says who gets a prompt). In a DM or a guest DM every message from
// the other party is a prompt, no `@` needed. In a group or a thread only a
// leading `@` is (§6.4); a reply in a thread is not a call. A person's `@` is
// always a prompt; an agent's only when the target's policy takes it, with
// one hop more than the prompt the agent is handling, and past the limit it
// degrades to a plain message with a system message saying so (rule 2,
// §6.5, §7.4). Pure: the queue items the caller then appends come out as
// drafts' essentials; the caller adds the venue words and the context pack.
import { parseSpeaker, personBehind, type Speaker } from "./identity.js";
import type { Message } from "./message.js";
import { parseMentions, resolveCalls, type RoleResolver, type RosterEntry } from "./mentions.js";
import { hopsOf, hopsSystemText, levelOf, overHops, type Level, type Relations } from "./sender.js";
import type { PromptFrom, Venue } from "./venue.js";

/** An agent present in the venue: its shown name, its speaker string, its owner's relations, and its own policy on whose `@` it takes (the venue's when none). */
export interface AgentInVenue { name: string; speaker: string; relations: Relations; policy?: { promptFrom: PromptFrom; allow: string[] } }

export interface DeliveryInput {
  venue: Venue;
  agents: AgentInVenue[];
  roster: RosterEntry[];
  resolve?: RoleResolver;
  /** When an agent speaks while handling a prompt: that prompt's hops. */
  handling?: number;
}

export interface PromptTo {
  agent: string;
  level: Level;
  hops: number;
  /** The role the call came through, when it did. */
  via?: string;
}

export interface Delivery {
  prompts: PromptTo[];
  /** Agents mentioned mid-sentence: a 「被提到」 on their board, no delivery. */
  mentioned: string[];
  /** People and 所有人 to notify. */
  notified: string[];
  /** Agents called but not prompted, and why. */
  degraded: { agent: string; reason: string }[];
  /** Roles nobody fits right now. */
  unresolved: string[];
  /** System messages the venue gets. */
  system: string[];
}

const NOTHING = (): Delivery => ({ prompts: [], mentioned: [], notified: [], degraded: [], unresolved: [], system: [] });

/** Whether the target takes an agent's `@` as a prompt (rule 2): its own policy, else the venue's; an allowlist names speakers, and naming a person names their agents. */
function takesAgentPrompt(speaker: Speaker & { kind: "agent" }, target: AgentInVenue, venue: Venue): boolean {
  const policy = target.policy ?? venue.policy;
  if (policy.promptFrom === "anyone") return true;
  if (policy.promptFrom === "nobody") return false;
  return policy.allow.includes(`agent:${speaker.name}@${speaker.owner}`) || policy.allow.includes(`person:${speaker.owner}`);
}

export function deliver(m: Message, d: DeliveryInput): Delivery {
  const out = NOTHING();
  const speaker = parseSpeaker(m.speaker);
  if (speaker === undefined || speaker.kind === "host" || m.kind === "system" || m.kind === "signal" || d.venue.kind === "channel") return out;
  const hops = speaker.kind === "agent" ? hopsOf(d.handling ?? 0) : 0;

  if (d.venue.kind === "dm" || d.venue.kind === "guest-dm") {
    const hostName = d.venue.host ?? d.venue.agents[0];
    const target = d.agents.find((a) => a.name === hostName);
    if (target === undefined || target.speaker === m.speaker) return out;
    // The owner cutting into a guest DM speaks to the guest, not to the agent (§6.2); in their own DM every word is a prompt.
    if (d.venue.kind === "guest-dm" && personBehind(speaker) === target.relations.owner) return out;
    out.prompts.push({ agent: target.name, level: levelOf(speaker, target.relations), hops });
    return out;
  }

  const parsed = parseMentions(m.text, d.roster);
  out.mentioned = parsed.mentions;
  out.notified = parsed.notified;
  const { calls, unresolved } = resolveCalls(parsed.calls, d.roster, d.resolve ?? (() => undefined));
  out.unresolved = unresolved;
  for (const call of calls) {
    const name = call.resolved ?? call.name;
    const target = d.agents.find((a) => a.name === name);
    if (target === undefined) { out.degraded.push({ agent: name, reason: "不在这个场所" }); continue; }
    if (target.speaker === m.speaker) continue;
    if (speaker.kind === "agent") {
      if (!takesAgentPrompt(speaker, target, d.venue)) { out.degraded.push({ agent: name, reason: "对方策略不收 agent 的指令，只当消息读到" }); continue; }
      const limit = d.venue.policy.hopLimit;
      if (overHops(hops, limit)) {
        out.degraded.push({ agent: name, reason: `超过 ${limit} 跳` });
        out.system.push(hopsSystemText(name, hops, limit));
        continue;
      }
    }
    const prompt: PromptTo = { agent: name, level: levelOf(speaker, target.relations), hops };
    if (call.resolved !== undefined) prompt.via = call.name;
    out.prompts.push(prompt);
  }
  return out;
}
```

```ts
// The 「需要我」 inbox (§6.9): one per person, made only of five kinds of
// thing — 定 (an ask for me or my role), 验收 (a deliverable or a card stage
// waiting on me), 放行 (a request for more than its sender may have), 卡住
// (my agent blocked, erring, refused again, or silent), 断了 (the desktop
// cannot reach the relay, or the daemon's heartbeat stopped). A new item
// notifies once, on the band, and to the phone bridge after the person has
// been away long enough; after that only an escalation notifies again. Mutes
// are by item, by venue, by kind. Everything else is silent and only on the
// board (rule 9). Pure: records in, rows and words out.

export type NeedKind = "offline" | "decide" | "accept" | "grant" | "stuck";

/** The order on the band and the board (§6.9): 断了 first, then 定, 验收, 放行, 卡住. */
export const NEED_ORDER: readonly NeedKind[] = ["offline", "decide", "accept", "grant", "stuck"];
export const NEED_VERBS: Readonly<Record<NeedKind, string>> = { offline: "断了", decide: "定", accept: "验收", grant: "放行", stuck: "卡住" };

/** The verb's colour on the band (§6.9): 放行 and 定 amber, 验收 green, 卡住 and 断了 red. */
export type Tone = "amber" | "green" | "red";
export const NEED_TONES: Readonly<Record<NeedKind, Tone>> = { decide: "amber", grant: "amber", accept: "green", stuck: "red", offline: "red" };

export interface Need {
  kind: NeedKind;
  /** Stable across renders: the thing it is about — an ask's message, a card, a grant request, an agent's session. */
  key: string;
  /** One sentence. */
  text: string;
  /** Button labels, in order; `defaultActions` gives each kind's. */
  actions: string[];
  since: string;
  venue?: string;
  /** How far it has escalated: 0 when new. A rise notifies again; nothing else does. */
  level: number;
}

export const WRITE_IN = "自己写";

/** Each kind's buttons (§6.9; the phone's extra 看演示 and the host's suggested fix from transport §7.2). */
export function defaultActions(kind: NeedKind, o: { options?: readonly string[]; settle?: number; suggestion?: string; phone?: boolean }): string[] {
  switch (kind) {
    case "decide": { const opts = (o.options ?? []).filter((x) => x !== WRITE_IN); return [...opts, WRITE_IN]; }
    case "accept": return [o.settle === undefined ? "接受" : `接受 · 结算 ${o.settle}`, "要改", ...(o.phone === true ? ["看演示"] : [])];
    case "grant": return ["允许一次", "允许此人", "拒绝"];
    case "stuck": return ["看原因", ...(o.suggestion === undefined ? [] : [o.suggestion]), "接管", "交回"];
    case "offline": return ["重试"];
  }
}

/** The items in the band's order: by kind, then oldest first; the input is left as it is. */
export function sortNeeds(items: readonly Need[]): Need[] {
  return [...items].sort((a, b) => NEED_ORDER.indexOf(a.kind) - NEED_ORDER.indexOf(b.kind) || (a.since < b.since ? -1 : a.since > b.since ? 1 : 0));
}

/** A mute (§6.9): by item, by venue, by kind, or by kind within a venue — every named field must match. */
export interface Mute { key?: string; venue?: string; kind?: NeedKind }

export function isMuted(n: Need, mutes: readonly Mute[]): boolean {
  return mutes.some((m) => (m.key === undefined || m.key === n.key) && (m.venue === undefined || m.venue === n.venue) && (m.kind === undefined || m.kind === n.kind));
}

/** What the person was last told about an item: its escalation then. */
export interface Seen { key: string; level: number }

/**
 * Which items to notify now (§6.9): one never told, or one escalated since
 * it was last told, unless muted. On the band whenever the person is here;
 * to the phone bridge only once they have been away longer than the
 * threshold.
 */
export function notifyNow(items: readonly Need[], seen: readonly Seen[], mutes: readonly Mute[], o: { awayMs: number; pushAfterMs: number }): { band: Need[]; phone: Need[] } {
  const told = new Map(seen.map((s) => [s.key, s.level] as const));
  const band = sortNeeds(items).filter((n) => !isMuted(n, mutes) && (told.get(n.key) ?? -1) < n.level);
  return { band, phone: o.awayMs > o.pushAfterMs ? band : [] };
}

/** One line on the band (§6.9): the verb and its colour, the sentence, the buttons, and how many more wait. */
export interface BandRow { verb: string; tone: Tone; text: string; actions: string[]; more: number }

export function bandRow(items: readonly Need[], mutes: readonly Mute[] = []): BandRow | undefined {
  const live = sortNeeds(items).filter((n) => !isMuted(n, mutes));
  const first = live[0];
  if (first === undefined) return undefined;
  return { verb: NEED_VERBS[first.kind], tone: NEED_TONES[first.kind], text: first.text, actions: first.actions, more: live.length - 1 };
}

/** The band when nothing needs the person (§6.9): the current card's state and whom it waits on, and whether the group has new messages that did not name them. */
export function quietLine(o: { card?: { id: string; state: string; waitingOn?: string }; unaddressed?: number }): string {
  const parts: string[] = [];
  if (o.card !== undefined) parts.push(`${o.card.id} ${o.card.state}${o.card.waitingOn === undefined ? "" : ` · 等${o.card.waitingOn}`}`);
  if (o.unaddressed !== undefined && o.unaddressed > 0) parts.push(`群里有 ${o.unaddressed} 条新消息，没点名你`);
  return parts.length === 0 ? "没有需要你的事" : parts.join(" · ");
}
```

```ts
// Presence and receipts (§6.10). A session's presence is one of five words;
// an agent account's is its most active session's. What others see of it is
// made only of what the venue may see: 「正在做 #42」 where #42 is visible,
// 「忙」 elsewhere. A person's message shows delivered or read-by; a prompt's
// three states are the queue's (queue.ts, senderStatus). Pure.

export type Presence = "thinking" | "tool" | "waiting" | "idle" | "asleep";

/** The owner's words for their own agent's state (the board's header). */
export const PRESENCE_WORDS: Readonly<Record<Presence, string>> = { thinking: "思考", tool: "跑工具", waiting: "等你", idle: "空闲", asleep: "睡着" };

const ACTIVITY: Readonly<Record<Presence, number>> = { thinking: 0, tool: 1, waiting: 2, idle: 3, asleep: 4 };

/** The most active of the account's sessions; none means asleep. */
export function agentPresence(sessions: readonly { presence: Presence }[]): Presence {
  let best: Presence = "asleep";
  for (const s of sessions) if (ACTIVITY[s.presence] < ACTIVITY[best]) best = s.presence;
  return best;
}

export const isAlive = (p: Presence): boolean => p !== "asleep";

/** The status others see (§6.10): the task only where this venue may see it, else 忙; the tool's seconds when known. */
export function presenceText(p: Presence, o: { task?: string; taskVisibleHere?: boolean; toolSeconds?: number }): string {
  const tool = `跑工具${o.toolSeconds === undefined ? "" : ` ${o.toolSeconds} 秒`}`;
  switch (p) {
    case "thinking": case "tool": {
      if (o.task !== undefined) return o.taskVisibleHere === true ? `正在做 ${o.task}${p === "tool" ? ` · ${tool}` : ""}` : "忙";
      return p === "tool" ? tool : "思考";
    }
    case "waiting": return "等主人";
    case "idle": return "空闲";
    case "asleep": return "睡着";
  }
}

/** The line under a person's message: who read it, or that it arrived, or that it is on its way (a pending block, transport §2.7). */
export function receiptLine(o: { delivered: boolean; readBy: readonly string[] }): string {
  if (!o.delivered) return "发送中";
  return o.readBy.length === 0 ? "送达" : `已读 · ${o.readBy.join("、")}`;
}
```
