# IM Plan E: The Share Card and the Guest Page

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The smallest spreading loop of the IM (main spec §1.3, §11; transport §7.1): a person makes a share card from their work, posts its address anywhere, and anyone who opens it reads the card and asks the author's agent without installing or registering — the question reaching the agent at guest level, the answer coming back onto the page.

**Architecture:** A share card is a delivery package without an acceptance mapping plus a public-scope declaration (§8.6, §11): one block in `cards/<id>.geml` under the owner, public to every reader (access rule), its id the relay's own (`/c/<卡>` names no owner). A guest's first word on a card makes a guest identity and a guest DM with the card's agent (§6.2); the card's public range becomes that DM's announce, so the agent reads it as the venue's context and the daemon writes it under the envelope. The guest page is one static HTML string served at `/c/<id>` to a browser and filled from three JSON calls, every node built with DOM calls (nothing the relay returns runs as markup); the words it shares with the board are rendered once in `src/core/im/share.ts` and tested there. The quota of §9.1 (访客问答上限) is counted on the relay per owner, per guest and per network address, until the ledger exists.

**Tech Stack:** TypeScript (`src/core/im` pure; `src/relay` Node), the relay's existing store and access rule, `node:test` over `dist/`.

**Before this plan:** plans A–D are on `main` (0.2.93): the core modules, the relay, the daemon, the Claude Code shell.

// The share card (§11, transport §7.1): a delivery package without an
// acceptance mapping (§8.6) plus a public-scope declaration, made by the
// owner from a session — the smallest spreading loop. This module is the
// record, the faults in what a client sends, the words the page and the
// board show, and the guest quota (§9.1's 访客问答上限 before the ledger
// exists). Pure: records in, records and words out.

/** 「怎么做出来的」: a milestone turn, an embed of that turn's record when given. */
export interface Replay { turn: number; label: string; src?: string }
/** 拿去用: the repository at a commit, the license, the AI statement. */
export interface UseTerms { repo?: string; commit?: string; license?: string; ai?: string }
export interface ShareInput {
  title: string;
  /** The one line (§11), and whether the agent drafted it (the owner edited it) or the owner typed it (rule 6). */
  summary: string;
  summarySource: "self" | "drafted";
  /** The agent that made it, by its shown name. */
  agent: string;
  turns?: number;
  /** A demo to play; a web build to open in the browser. */
  demo?: string;
  open?: string;
  replay: Replay[];
  /** The public range: what the agent may talk about; the rest it says it does not. */
  scope: string;
  /** Evidence lines the host observed. */
  evidence: string[];
  use: UseTerms;
}
/** The card as the relay stores it: public to every reader, dated, the id the relay drew. */
export interface ShareCard extends ShareInput { kind: "share"; id: string; public: true; at: string }

export const TITLE_LIMIT = 200;
export const SUMMARY_LIMIT = 1000;
export const SCOPE_LIMIT = 500;
export const REPLAY_LIMIT = 12;
const URL_RE = /^https?:\/\/\S+$/;

const str = (v: unknown): string | undefined => (typeof v === "string" ? v : undefined);
const obj = (v: unknown): Record<string, unknown> | undefined => (typeof v === "object" && v !== null && !Array.isArray(v) ? (v as Record<string, unknown>) : undefined);
const strings = (v: unknown): string[] => (Array.isArray(v) ? v.filter((x): x is string => typeof x === "string" && x.trim() !== "").map((x) => x.trim()) : []);

/** The faults in what a client sent, as a person reads them; none means `shareInput` gives the record. */
export function shareFaults(v: unknown): string[] {
  const o = obj(v);
  if (o === undefined) return ["要一个 JSON 对象"];
  const out: string[] = [];
  const title = str(o["title"])?.trim() ?? "";
  if (title === "") out.push("缺标题");
  else if ([...title].length > TITLE_LIMIT) out.push(`标题最多 ${TITLE_LIMIT} 字`);
  if ((str(o["agent"])?.trim() ?? "") === "") out.push("缺 agent 的名字");
  const scope = str(o["scope"])?.trim() ?? "";
  if (scope === "") out.push("缺公开范围");
  else if ([...scope].length > SCOPE_LIMIT) out.push(`公开范围最多 ${SCOPE_LIMIT} 字`);
  if ([...(str(o["summary"]) ?? "")].length > SUMMARY_LIMIT) out.push(`一句话最多 ${SUMMARY_LIMIT} 字`);
  const source = o["summarySource"];
  if (source !== undefined && source !== "self" && source !== "drafted") out.push("一句话的来源是 self 或 drafted");
  const turns = o["turns"];
  if (turns !== undefined && !(Number.isInteger(turns) && (turns as number) >= 0)) out.push("轮数要是非负整数");
  for (const k of ["demo", "open"] as const) { const u = str(o[k]); if (o[k] !== undefined && (u === undefined || !URL_RE.test(u))) out.push(`${k} 要是 http(s) 地址`); }
  const replay = o["replay"];
  if (replay !== undefined) {
    if (!Array.isArray(replay)) out.push("回放要是数组");
    else if (replay.length > REPLAY_LIMIT) out.push(`回放最多 ${REPLAY_LIMIT} 个里程碑`);
    else for (const r of replay) {
      const e = obj(r);
      if (e === undefined || !Number.isInteger(e["turn"]) || (e["turn"] as number) < 1 || (str(e["label"])?.trim() ?? "") === "") { out.push("回放的每一项要有轮次（正整数）和标签"); break; }
    }
  }
  if (o["use"] !== undefined && obj(o["use"]) === undefined) out.push("拿去用要是对象");
  return out;
}

/** The record from a body `shareFaults` passed: trimmed, defaults filled, nothing else kept. */
export function shareInput(v: unknown): ShareInput | undefined {
  if (shareFaults(v).length > 0) return undefined;
  const o = obj(v)!;
  const use = obj(o["use"]) ?? {};
  const input: ShareInput = { title: str(o["title"])!.trim(), summary: (str(o["summary"]) ?? "").trim(), summarySource: o["summarySource"] === "drafted" ? "drafted" : "self", agent: str(o["agent"])!.trim(), replay: [], scope: str(o["scope"])!.trim(), evidence: strings(o["evidence"]), use: {} };
  if (o["turns"] !== undefined) input.turns = o["turns"] as number;
  for (const k of ["demo", "open"] as const) { const u = str(o[k]); if (u !== undefined) input[k] = u; }
  for (const r of Array.isArray(o["replay"]) ? o["replay"] : []) {
    const e = obj(r)!;
    const item: Replay = { turn: e["turn"] as number, label: str(e["label"])!.trim() };
    const src = str(e["src"]); if (src !== undefined) item.src = src;
    input.replay.push(item);
  }
  for (const k of ["repo", "commit", "license", "ai"] as const) { const s = str(use[k])?.trim(); if (s !== undefined && s !== "") input.use[k] = s; }
  return input;
}

/** A stored card block read back under the document's id (the block's own `id` is the block's); undefined when it is no share card. */
export function readShare(v: unknown, id: string): ShareCard | undefined {
  const o = obj(v);
  if (o === undefined || o["kind"] !== "share" || o["public"] !== true) return undefined;
  const at = str(o["at"]);
  const input = shareInput(o);
  return at === undefined || input === undefined ? undefined : { kind: "share", id, public: true, at, ...input };
}

/** 「John 的作品 · xiaoping 做的 · 63 轮 · 2026-10-30 · 有凭据」 */
export function byline(c: ShareCard, owner: string): string {
  return [`${owner} 的作品`, `${c.agent} 做的`, ...(c.turns !== undefined ? [`${c.turns} 轮`] : []), c.at.slice(0, 10), ...(c.evidence.length > 0 ? ["有凭据"] : [])].join(" · ");
}
/** The tag under the one line (rule 6): the agent drafted it and the owner edited it; nothing when the owner typed it. */
export const summaryTag = (c: ShareCard, owner: string): string | undefined => (c.summarySource === "drafted" ? `${c.agent} 起草 · ${owner} 改过` : undefined);
/** The public range as the card states it, with what the agent does about the rest. */
export const rangeWords = (c: ShareCard): string => `${c.scope} · 其余不在此，${c.agent} 不会讲`;
/** 拿去用 in one line: the repository at its commit, the license, the AI statement; nothing when the card says none of it. */
export function useLine(c: ShareCard): string | undefined {
  const parts = [...(c.use.repo !== undefined ? [`仓库 ${c.use.repo}${c.use.commit !== undefined ? ` @ ${c.use.commit}` : ""}`] : []), ...(c.use.license !== undefined ? [`许可：${c.use.license}`] : []), ...(c.use.ai !== undefined ? [c.use.ai] : [])];
  return parts.length === 0 ? undefined : parts.join(" · ");
}
/** The right column's header: 「你是访客，不用注册 · 今天还能问 3 句 · 这几句的 token John 请客」 */
export const guestHeader = (left: number, owner: string): string => `你是访客，不用注册 · 今天还能问 ${left} 句 · 这几句的 token ${owner} 请客`;
/** The badge under the agent's answer on the page: 「xiaoping · 代 John · 只讲公开范围内的」 */
export const agentBadge = (agent: string, owner: string): string => `${agent} · 代 ${owner} · 只讲公开范围内的`;
/** The line the daemon writes under a guest's envelope (§11): the public range, and what to do outside it. */
export const scopeLine = (scope: string): string => `[公开范围：${scope}；不在范围内的明说不讲，请访客给主人留言]`;

/** The owner's guest quota (§9.1): questions a day across all guests, a guest's a day, a network address's a day. */
export interface GuestQuota { perDay: number; perGuestPerDay: number; perIpPerDay: number }
export const DEFAULT_GUEST_QUOTA: GuestQuota = { perDay: 50, perGuestPerDay: 3, perIpPerDay: 20 };
/** A quota from a body or a block: each a positive integer, else the default. */
export function quotaOf(v: unknown): GuestQuota {
  const o = obj(v) ?? {};
  const n = (k: keyof GuestQuota): number => { const x = o[k]; return Number.isInteger(x) && (x as number) > 0 ? (x as number) : DEFAULT_GUEST_QUOTA[k]; };
  return { perDay: n("perDay"), perGuestPerDay: n("perGuestPerDay"), perIpPerDay: n("perIpPerDay") };
}
/** Why a question is refused when a quota is used up. */
export const QUOTA_WORDS = { owner: "作者今天的访客额度用完了，明天再来", guest: "你今天的几句问完了，明天再来", ip: "这个网络今天问得太多了，明天再来" } as const;
/** Whether two instants fall on the same UTC day. */
export const sameDay = (a: string, b: string): boolean => a.slice(0, 10) === b.slice(0, 10);

  /** The card block of `cards/<id>.geml`, when it is a share card. */
  private readShareDoc(owner: string, id: string): ShareCard | undefined {
    if (!NAME.test(owner) || !NAME.test(id)) return undefined;
    return readShare(readDoc(this.root, owner, `cards/${id}.geml`).blocks.find((b) => b.id === "card")?.value, id);

  /** Whose a card is: ids are the relay's (`/c/<卡>` names no owner), so the owners are scanned. */
  private shareOwner(id: string): string | undefined {
    return NAME.test(id) ? listOwners(this.root).find((o) => o !== RELAY_OWNER && this.readShareDoc(o, id) !== undefined) : undefined;
  }

  /** A share card under the owner's name (§11): a delivery package without an acceptance mapping plus the public range; the agent must be theirs; the id is drawn here. */
  createShare(by: TokenRecord, body: unknown): Result<{ card: ShareCard; url: string }> {
    const s = parseSpeaker(by.speaker);
    if (by.scope !== "person" || s?.kind !== "person") return fail(403, "只有本人的令牌能发分享卡");
    const faults = shareFaults(body);
    if (faults.length > 0) return fail(400, faults.join("；"));
    const input = shareInput(body)!;
    if (!readDoc(this.root, s.id, `agents/${input.agent}.geml`).exists) return fail(400, `没有这个 agent：${input.agent}`);
    for (let tries = 0; tries < 16; tries++) {
      const id = `c-${randomBytes(2).toString("hex")}`;
      if (this.shareOwner(id) !== undefined) continue;
      const doc = `cards/${id}.geml`;
      const a = appendSealed(this.root, s.id, doc, "card", { kind: "share", public: true, at: this.now(), ...input } as unknown as Record<string, JsonValue>);
      this.emit(s.id, doc, a.block);
      return { ok: true, card: this.readShareDoc(s.id, id)!, url: `/c/${id}` };
    }
    return fail(500, "卡号都被占了");
  }

  /** A person's share cards, newest first. */
  sharesOf(person: string): ShareCard[] {
    return listDocs(this.root, person, "cards").map((d) => this.readShareDoc(person, d.slice("cards/".length, -".geml".length))).filter((c): c is ShareCard => c !== undefined).sort((a, b) => (a.at < b.at ? 1 : a.at > b.at ? -1 : 0));
  }

  /** The owner's guest quota (§9.1): the last `quota-N` block on their page, else the defaults. */
  guestQuotaOf(owner: string): GuestQuota {
    const all = readDoc(this.root, owner, this.personDoc(owner)).blocks.filter((b) => /^quota-\d+$/.test(b.id));
    return quotaOf(all[all.length - 1]?.value);
  }

  setGuestQuota(by: TokenRecord, body: unknown): Result<{ quota: GuestQuota }> {
    const s = parseSpeaker(by.speaker);
    if (by.scope !== "person" || s?.kind !== "person") return fail(403, "只有本人能设访客额度");
    const quota = quotaOf(body);
    const doc = this.personDoc(s.id);
    const a = appendBlock(this.root, s.id, doc, nextId("quota"), (_d, prev, bid) => sealed({ id: bid, ...quota, at: this.now() }, prev));
    this.emit(s.id, doc, a.block);
    return { ok: true, quota };
  }

  /** The guests' questions today in the owner's guest DMs: all guests', or one guest's. */
  guestQuestionsToday(owner: string, now: string, guest?: string): number {
    let n = 0;
    for (const vd of this.venuesOf(owner)) {
      if (vd.venue.kind !== "guest-dm") continue;
      for (const m of vd.messages) if (m.speaker.startsWith("guest:") && (guest === undefined || m.speaker === guest) && sameDay(m.at, now)) n++;
    }
    return n;
  }

  /** What the page shows (transport §7.1); `guest` is the viewer's speaker when they sent a guest token. */
  shareOf(id: string, guest?: string): Result<ShareView> {
    const owner = this.shareOwner(id);
    const card = owner === undefined ? undefined : this.readShareDoc(owner, id);
    if (owner === undefined || card === undefined) return fail(404, "没有这张卡");
    const now = this.now();
    const quota = this.guestQuotaOf(owner);
    const presence = this.presenceOf(`agent:${card.agent}@${owner}`).presence;
    const ownerLeft = Math.max(0, quota.perDay - this.guestQuestionsToday(owner, now));
    const left = guest === undefined ? quota.perGuestPerDay : Math.max(0, Math.min(ownerLeft, quota.perGuestPerDay - this.guestQuestionsToday(owner, now, guest)));
    const tag = summaryTag(card, owner), use = useLine(card);
    const view: ShareView = {
      owner, card, agent: { name: card.agent, presence, presenceWord: presenceText(presence, {}) },
      quota: { perGuestPerDay: quota.perGuestPerDay, ownerLeftToday: ownerLeft },
      lines: { byline: byline(card, owner), ...(tag !== undefined ? { summaryTag: tag } : {}), range: rangeWords(card), ...(use !== undefined ? { use } : {}), guestHeader: guestHeader(left, owner), agentBadge: agentBadge(card.agent, owner) },
    };
    if (guest !== undefined) {
      const venue = guestVenueId(id, guest);
      view.guest = { speaker: guest, left, ...(this.venueOf(owner, venue) !== undefined ? { venue } : {}) };
    }
    return { ok: true, ...view };
  }

  /** A guest's question on a card (transport §7.1): a guest identity when none was sent, their guest DM with the card's agent at the first word (the public range its announce), the question appended as the guest and delivered at guest level — within the network address's, the owner's and the guest's quota for the day. A question sent again under its id is answered again, not counted again. */
  askAsGuest(by: TokenRecord | undefined, cardId: string, body: unknown, ip?: string): Result<{ token?: string; speaker: string; venue: string; message: Message; left: number }> {
    const owner = this.shareOwner(cardId);
    const card = owner === undefined ? undefined : this.readShareDoc(owner, cardId);
    if (owner === undefined || card === undefined) return fail(404, "没有这张卡");
    if (by !== undefined && by.scope !== "guest") return fail(403, "这条路只给访客；成员请在场所里说");
    const b = obj(body);
    if (b === undefined) return fail(400, "要一个 JSON 对象");
    const text = (str(b["text"]) ?? "").trim();
    if (text === "") return fail(400, "正文为空");
    const now = this.now();
    const quota = this.guestQuotaOf(owner);
    if (ip !== undefined && this.ipToday(ip, now) >= quota.perIpPerDay) return fail(429, QUOTA_WORDS.ip);
    if (this.guestQuestionsToday(owner, now) >= quota.perDay) return fail(429, QUOTA_WORDS.owner);
    const issued = by === undefined ? issueToken(this.root, { speaker: `guest:g-${rand()}`, scope: "guest", by: "guest", at: now }) : undefined;
    const guest = issued?.record ?? by!;
    const venueId = guestVenueId(cardId, guest.speaker);
    const used = this.guestQuestionsToday(owner, now, guest.speaker);
    const mid = str(b["id"]);
    const existing = mid === undefined ? undefined : this.venueOf(owner, venueId)?.messages.find((m) => m.id === mid);
    if (existing !== undefined) return { ok: true, speaker: guest.speaker, venue: venueId, message: existing, left: Math.max(0, quota.perGuestPerDay - used) };
    if (used >= quota.perGuestPerDay) return fail(429, QUOTA_WORDS.guest);
    if (this.venueOf(owner, venueId) === undefined) {
      const made = this.createVenueAs(owner, { id: venueId, kind: "guest-dm", name: shownName(guest.speaker), people: [guest.speaker], agents: [`agent:${card.agent}@${owner}`], announce: card.scope, card: cardId });
      if (!made.ok) return made;
    }
    const r = this.appendMessage(guest, owner, venueId, { id: mid ?? `m-${rand()}`, text });
    if (!r.ok) return r;
    if (ip !== undefined) this.ipBump(ip, now);
    return { ok: true, ...(issued !== undefined ? { token: issued.token } : {}), speaker: guest.speaker, venue: venueId, message: r.message, left: Math.max(0, quota.perGuestPerDay - used - 1) };
  }

  /** Questions by network address today, in memory: a relay restarted forgets them, which errs on the guests' side. */
  private readonly ips = new Map<string, { day: string; n: number }>();
  private ipToday(ip: string, now: string): number { const e = this.ips.get(ip); return e !== undefined && sameDay(e.day, now) ? e.n : 0; }
  private ipBump(ip: string, now: string): void { this.ips.set(ip, { day: now.slice(0, 10), n: this.ipToday(ip, now) + 1 }); }
