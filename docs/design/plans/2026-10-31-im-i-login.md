# IM Plan I: Login, Registration, the Hosted Agent's Account, the Pages

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Transport §7.1's other roads and §2.5's login: `/login` by email code (登录即注册), the first login naming the person and giving them a hosted agent (transport §6: an account, its token kept for the platform's runner, its first word in a DM — the runner itself is plan L); the claim pages `/g/<礼物>` and `/r/<红包>`; 打赏 on the share card once logged in, with the 登录 and 要一个自己的 agent buttons live; the mail route the operator chooses (a file by default, a command with `--mail-command`).

**Architecture:** `src/core/im/login.ts` (pure) holds the rules and the words: the email and name checks, the reserved names, the code's shape and life, the mail's text, the welcome. The relay keeps codes (hashed, like tokens) and the email→person map under its own owner `_relay`, sends through a `Mailer` it is given (`src/relay/mail.ts`: `fileMailer`, `commandMailer`), and on a first login writes the person, their hosted agent and the DM. `src/relay/page.ts` grows from one page to four over one shared script (fetch with a token, the stored person, the login box), still DOM-built. The API gets `POST /login` and `POST /login/verify` on the token-free side; the server serves the pages by path.

**Tech Stack:** TypeScript (`src/core` pure, `src/relay`), `node:test` over `dist/`.

**Before this plan:** plans A–H are on `main` (0.3.1).

```ts
// Login and registration (transport §2.5 登录, §7.1 /login): a person is
// their email; a six-digit code mailed to it is the proof, good for ten
// minutes; the first login names the person and gives them a hosted agent
// (transport §6) whose first word waits in a DM. Pure: the rules and the
// words; the relay keeps the codes and sends the mail.

export const HOSTED_AGENT = "helper";
export const CODE_TTL_MS = 10 * 60_000;
export const CODES_PER_EMAIL = 3;
export const CODES_PER_IP_HOUR = 10;
export const CODE_RE = /^\d{6}$/;
/** The API's roads and the speakers' kinds, which no person may be named after. */
export const RESERVED_NAMES: ReadonlySet<string> = new Set(["me", "cards", "tasks", "tokens", "venues", "agents", "queues", "presence", "inbox", "trust", "grants", "quota", "wallet", "guests", "c", "ask", "g", "r", "login", "subscribe", "relay", "host", "platform", "guest", "person", "agent"]);
const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const NAME_RE = /^[\p{L}\p{N}][\p{L}\p{N}_-]{0,63}$/u;

export const normalizeEmail = (v: string): string => v.trim().toLowerCase();

export function emailFault(v: unknown): string | undefined {
  return typeof v === "string" && EMAIL_RE.test(normalizeEmail(v)) ? undefined : "要一个邮箱地址";
}

export function nameFault(v: unknown): string | undefined {
  if (typeof v !== "string" || v.trim() === "") return "新用户要取个名字";
  if (!NAME_RE.test(v)) return "名字不合法：字母、数字、_ 和 -，最多 64 个";
  if (RESERVED_NAMES.has(v.toLowerCase())) return "这个名字留给系统";
  return undefined;
}

export const codeMail = (code: string): { subject: string; text: string } => ({ subject: "Gemsbang 登录验证码", text: `你的验证码是 ${code}，10 分钟内有效。不是你在登录的话，不用管这封信。` });

  return `你好 ${person}，我是 ${HOSTED_AGENT}，你的托管 agent，代你守在这里。想让我知道你的项目，就在电脑上运行 gemworks connect，把会话接过来；有人在你的分享卡上提问，我会按公开范围答，答不了的给你留言。`;
```

```ts
// Where a login code goes (transport §7.1 /login): the operator's choice.
// By default a file under the relay's own folder, for whatever routes mail
// on that box; `gemworks relay --mail-command` runs a command with the
// address and the subject in its environment and the text on stdin
// (sendmail -t, a curl to a provider). The relay never waits on it.
import { spawn } from "node:child_process";
import { randomBytes } from "node:crypto";
import { mkdirSync, writeFileSync } from "node:fs";
import { join } from "node:path";

export interface Mail { to: string; subject: string; text: string }
export type Mailer = (m: Mail) => void | Promise<void>;

/** Each mail a file `<root>/_relay/mail/<stamp>-<rand>.txt`: the headers, a blank line, the text. */
export function fileMailer(root: string): Mailer {
  return (m) => {
    const dir = join(root, "_relay", "mail");
    mkdirSync(dir, { recursive: true });
    writeFileSync(join(dir, `${Date.now()}-${randomBytes(3).toString("hex")}.txt`), `To: ${m.to}\nSubject: ${m.subject}\n\n${m.text}\n`);
  };
}

/** The command through the shell, MAIL_TO and MAIL_SUBJECT in its environment, the text on stdin; a non-zero exit rejects with its stderr. */
export function commandMailer(command: string): Mailer {
  return (m) => new Promise<void>((done, failed) => {
    const win = process.platform === "win32";
    const child = spawn(win ? "cmd.exe" : "sh", win ? ["/d", "/s", "/c", command] : ["-c", command], { env: { ...process.env, MAIL_TO: m.to, MAIL_SUBJECT: m.subject }, stdio: ["pipe", "ignore", "pipe"] });
    let err = "";
    child.stderr.on("data", (c: Buffer) => { err += c; });
    child.on("error", failed);
    child.on("exit", (code) => { if (code === 0) done(); else failed(new Error(`mail command exited ${code}: ${err.trim()}`)); });
    child.stdin.end(m.text);
  });
}
```

```ts
    const code = String(100_000 + (randomBytes(4).readUInt32BE(0) % 900_000));
    appendSealed(this.root, RELAY_OWNER, LOGINS, `login-${rand()}`, { email, code: hashToken(code), at: now, ...(ip !== undefined ? { ip } : {}) });
    const m = codeMail(code);
    try { void Promise.resolve(this.mailer({ to: email, subject: m.subject, text: m.text })).catch(() => { /* the operator's route failed; the next code tries again */ }); } catch { /* the same */ }
    return { ok: true, sent: true, known: this.personOfEmail(email) !== undefined };
  }

  /** The code checked (§7.1): a person's token for the email's person; a first login names them, writes their page, gives them the hosted agent (transport §6) whose first word waits in their DM, and keeps the agent's token for the platform's runner. */
  finishLogin(body: unknown): Result<{ token: string; person: string; new: boolean; agent?: string; venue?: string }> {
    const b = obj(body) ?? {};

    if (fault !== undefined) return fail(400, fault);

    const login = [...d.blocks].reverse().find((x) => x.id.startsWith("login-") && x.value["email"] === email && x.value["code"] === h && !used.has(x.id) && t - Date.parse(str(x.value["at"]) ?? "") < CODE_TTL_MS);
    if (login === undefined) return fail(401, LOGIN_WORDS.wrong);
    const known = this.personOfEmail(email);
    if (known !== undefined) {
      appendSealed(this.root, RELAY_OWNER, LOGINS, `used-${rand()}`, { login: login.id, at: now });
      return { ok: true, token: issueToken(this.root, { speaker: `person:${known}`, scope: "person", by: "login", at: now }).token, person: known, new: false };
    }
    const nf = nameFault(b["name"]);
    if (nf !== undefined) return fail(400, nf);
    const name = b["name"] as string;
    if (this.hasPage(name)) return fail(409, LOGIN_WORDS.taken);
    appendSealed(this.root, RELAY_OWNER, LOGINS, `used-${rand()}`, { login: login.id, at: now });
    const person = this.issuePerson(name);
    if (!person.ok) return person;
    appendSealed(this.root, RELAY_OWNER, EMAILS, `email-${rand()}`, { email, person: name, at: now });
    const agent = this.issueAgent(person.record, HOSTED_AGENT);
    if (!agent.ok) return agent;
    appendSealed(this.root, RELAY_OWNER, HOSTED, `hosted-${rand()}`, { person: name, agent: HOSTED_AGENT, token: agent.token, at: now });
    const venue = `dm-${HOSTED_AGENT}`;
    const made = this.createVenueAs(name, { id: venue, kind: "dm", name: HOSTED_AGENT, people: [], agents: [agent.record.speaker] });
    if (!made.ok) return made;
    this.appendMessage(agent.record, name, venue, { id: "m-welcome", text: welcomeText(name) });
    return { ok: true, token: person.token, person: name, new: true, agent: HOSTED_AGENT, venue };
  }

  /** The hosted agents with the tokens the platform's runner (plan L) drives them with: kept in the clear under the relay's own owner, which no person can read. */
  hostedAgents(): { person: string; agent: string; token: string }[] {
    return readDoc(this.root, RELAY_OWNER, HOSTED).blocks.filter((x) => x.id.startsWith("hosted-")).map((x) => ({ person: str(x.value["person"]) ?? "", agent: str(x.value["agent"]) ?? "", token: str(x.value["token"]) ?? "" }));
  }
```
