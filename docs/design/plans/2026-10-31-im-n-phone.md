# IM Plan N: The Phone Bridge — the Inbox on Telegram

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The phone bridge (transport §7.2): a bot on the person's phone — Telegram first, 飞书 later through the same interface — bound with a one-time code made on the desktop, holding an inbox-scope token of the person and nothing more; it mirrors the inbox's five kinds of need as small cards with inline buttons, a press (or a typed reply to a card) flowing back through the relay as the person's own answer (human 来源); `/tell <agent> <text>` goes into that agent's queue at the owner's level, first in line, and the bot says so in the spec's words; the bot's first line is fixed (「只镜像需要你的事 · 群里的话不在这里」); no group talk, no bot-to-bot, no files.

**Architecture:** The relay learns the binding (`phoneCode` for the person, `phoneBind` for the bot: a six-digit code, ten minutes, spent on use, an inbox-scope token issued), `answerNeed` (one dispatcher by the need's key — an ask's reply, a grant or refusal, an allowance, a digest's 定, a match's nod, a card's acceptance or changes, 看原因 and 交回 on a 卡住 — acting as the person for an inbox-scope token) and `tell` (the person's word into their DM with the agent, which the ordinary delivery turns into an owner-level prompt). `src/phone/bridge.ts` holds the `Bridge` (bindings under its home, the mirror loop, the handlers) over a `Transport` interface; `src/phone/telegram.ts` is the Telegram transport (long polling, inline keyboards, callback answers); the tests drive a scripted transport and a fake fetch. The CLI gets `gemworks phone` and `connect phone-code`.

**Tech Stack:** TypeScript (`src/relay`, `src/phone`, `src/cli.ts`), `node:test` over `dist/`.

**Before this plan:** plans A–M are on `main` (0.3.6).

```ts
  // --- the phone bridge (transport §7.2) ---

  /** A one-time code for the bot (绑定): six digits, ten minutes, its hash kept under the relay's own name. */
  phoneCode(by: TokenRecord): Result<{ code: string; expires: string }> {
    const s = parseSpeaker(by.speaker);
    if (by.scope !== "person" || s?.kind !== "person") return fail(403, "只有本人能生成绑定码");
    const code = String(100_000 + (randomBytes(4).readUInt32BE(0) % 900_000));
    const at = this.now();
    appendSealed(this.root, RELAY_OWNER, PHONE, `code-${rand()}`, { person: s.id, code: hashToken(code), at });
    return { ok: true, code, expires: new Date(Date.parse(at) + CODE_TTL_MS).toISOString().replace(/\.\d{3}Z$/, "Z") };
  }

  /** The bot binds with the code: an inbox-scope token of the person — the inbox's reads and answers, a word to one's own agent, nothing more — and the code is spent. */
  phoneBind(body: unknown): Result<{ token: string; person: string }> {
    const code = (str(obj(body)?.["code"]) ?? "").trim();
    if (!/^\d{6}$/.test(code)) return fail(400, "绑定码是六位数字");
    const d = readDoc(this.root, RELAY_OWNER, PHONE);
    const spent = new Set(d.blocks.filter((b) => b.id.startsWith("bound-")).map((b) => str(b.value["code"])));
    const now = this.now();
    const hit = [...d.blocks].reverse().find((b) => b.id.startsWith("code-") && !spent.has(b.id) && str(b.value["code"]) === hashToken(code) && Date.parse(str(b.value["at"]) ?? "") + CODE_TTL_MS > Date.parse(now));
    if (hit === undefined) return fail(404, "绑定码不对或过期了");
    const person = str(hit.value["person"]) ?? "";
    const issued = issueToken(this.root, { speaker: `person:${person}`, scope: "inbox", by: `phone:${hit.id}`, at: now });
    appendSealed(this.root, RELAY_OWNER, PHONE, `bound-${rand()}`, { code: hit.id, person, at: now });
    return { ok: true, token: issued.token, person };
  }

  /** An inbox-scope token acting as the person (transport §7.2 只限收件箱动作的本人令牌), for the answers below and nothing else. */
  private asPerson(by: TokenRecord): TokenRecord | undefined {
    const s = parseSpeaker(by.speaker);
    return (by.scope === "person" || by.scope === "inbox") && s?.kind === "person" ? { ...by, scope: "person" } : undefined;
  }

  /** A need answered by its key (§6.9), as the band and the phone do it: an ask's reply into its venue; 放行 on a request or an agent's ask for credits; 定 on a digest or a match; 验收 on a card (接受, 要改 with the words, 看演示 is the card's address back); on a 卡住, 看原因 is words back and 交回 gives the card back. The answer is the person's own (human 来源). */
  answerNeed(by: TokenRecord, key: string, action: string, text?: string): Result<{ said?: string }> {
    const me = this.asPerson(by);
    if (me === undefined) return fail(403, "只有本人的令牌能答复");
    const a = parseAddress(key);
    if (a === undefined || a.block === undefined || a.relay !== this.name) return fail(400, `不认识这件事：${key}`);
    const words = (text ?? "").trim();
    const venue = /^venues\/([^/]+)\.geml$/.exec(a.doc);
    if (venue !== null) {
      const reply = action === WRITE_IN ? words : action;
      if (reply === "") return fail(400, "自己写：缺正文");
      const r = this.appendMessage(me, a.owner, venue[1]!, { id: `m-${String(Date.parse(this.now())).padStart(13, "0")}-${rand()}`, text: reply, replyTo: a.block });
      return r.ok ? { ok: true, said: reply } : r;
    }
    if (/^digests\/[^/]+\.geml$/.test(a.doc)) {
      if (action === WRITE_IN && words === "") return fail(400, "自己写：缺正文");
      const r = this.answerDigest(me, { draft: a.block, publish: action !== "不发", ...(action === WRITE_IN ? { text: words } : {}) });
      return r.ok ? { ok: true } : r;
    }
    if (/^matches\/[^/]+\.geml$/.test(a.doc)) {
      const r = this.answerMatch(me, a.owner, a.doc.slice("matches/".length, -".geml".length), { yes: action === "亮名片" });
      return r.ok ? { ok: true } : r;
    }
    if (/^people\/[^/]+\.geml$/.test(a.doc)) {
      if (a.block.startsWith("cap-ask-")) { const r = this.allowCredits(me, { ask: a.block, allowed: action !== "不行" }); return r.ok ? { ok: true } : r; }
      const request = readDoc(this.root, a.owner, a.doc).blocks.find((b) => b.id === a.block);
      const from = str(request?.value["from"]), block = str(request?.value["block"]);
      if (from === undefined || block === undefined) return fail(404, "没有这个申请");
      if (action === "拒绝") { const r = this.refuse(me, a.block); return r.ok ? { ok: true } : r; }
      if (action !== "允许一次" && action !== "允许此人") return fail(400, `放行没有这个答复：${action}`);

    const me = this.asPerson(by);
    if (me === undefined) return fail(403, "只有本人的令牌能说话");
    const person = me.speaker.slice("person:".length);
    const b = obj(body) ?? {};
    const agent = str(b["agent"]) ?? "", text = (str(b["text"]) ?? "").trim();
    if (!NAME.test(agent) || !this.agentsOf(person).includes(agent)) return fail(404, "没有这个 agent");
    if (text === "") return fail(400, "缺正文");
    const id = `dm-${agent}`;
    if (this.venueOf(person, id) === undefined) {
      const made = this.createVenueAs(person, { id, kind: "dm", name: agent, agents: [`agent:${agent}@${person}`] });
      if (!made.ok) return made;
    }
    const r = this.appendMessage(me, person, id, { id: `m-${String(Date.parse(this.now())).padStart(13, "0")}-${rand()}`, text });
    return r.ok ? { ok: true, venue: `${person}/${id}`, agent } : r;
  }
```

```ts
// The phone bridge (transport §7.2): a bot on the person's phone that
// mirrors the inbox's five kinds of need as small cards with buttons — a
// press, or a typed reply to a card, flows back through the relay as the
// person's own answer — and takes /tell to one's own agent. It binds with a
// one-time code from the desktop, holds an inbox-scope token only, and
// mirrors nothing else: no group talk, no bot-to-bot, no files. The
// transport (Telegram first, 飞书 next) is a small interface; a scripted one
// drives the tests.
import { existsSync, readdirSync, readFileSync } from "node:fs";
import { join } from "node:path";
import { NEED_VERBS, type Need } from "../core/im/needs-me.js";
import { RelayClient, reasonOf } from "../daemon/client.js";
import { writeAtomic } from "../host-fs.js";

export interface Button { label: string; data: string }
/** What a bot platform must do: hand over texts (with the id of the message they reply to, when any) and button presses, each with its chat; send a message with rows of buttons and give back its id. */
export interface Transport {
  start(h: { onText: (chat: string, text: string, replyTo?: string) => void | Promise<void>; onPress: (chat: string, data: string) => void | Promise<void> }): void;
  stop(): void;
  send(chat: string, text: string, buttons?: Button[][]): Promise<string>;
}
/** One chat's binding: whose inbox, the token, what was shown (by key), the cards by their number (for the buttons' data), the last number. */
export interface Binding { person: string; token: string; seen: string[]; cards: Record<string, { key: string; actions: string[]; message: string }>; n: number }

export const FIRST_LINE = "只镜像需要你的事 · 群里的话不在这里";
export const BIND_FIRST = "先绑定：在桌面端或网页生成一次性码，然后发 /bind <码>";
export const HELP_LINE = "只认 /tell <agent> <文字>，和对一张卡的回复";
/** The bot's word after /tell (transport §7.2). */
export const tellReply = (agent: string, person: string): string => `已进 ${agent} 的队列，主人等级，排第一。它会在群里转述你这句，标『${person} 原话』`;
const SEEN_LIMIT = 500;

export interface BridgeOptions { relayUrl: string; home: string; transport: Transport; pollMs?: number; log?: (line: string) => void; fetchImpl?: typeof fetch }

export class Bridge {
  private readonly relayUrl: string;
  private readonly home: string;
  private readonly transport: Transport;
  private readonly pollMs: number;
  private readonly log: (line: string) => void;
  private readonly fetchImpl: typeof fetch | undefined;
  private timer: ReturnType<typeof setInterval> | undefined;

  constructor(o: BridgeOptions) {
    this.relayUrl = o.relayUrl;
    this.home = o.home;
    this.transport = o.transport;
    this.pollMs = o.pollMs ?? 15_000;
    this.log = o.log ?? (() => {});
    this.fetchImpl = o.fetchImpl;
  }

  start(): void {
    this.transport.start({ onText: (chat, text, replyTo) => this.onText(chat, text, replyTo), onPress: (chat, data) => this.onPress(chat, data) });
    this.timer = setInterval(() => { void this.mirror().catch((e: unknown) => this.log(e instanceof Error ? e.message : String(e))); }, this.pollMs);
    this.timer.unref?.();
  }

  stop(): void {
    if (this.timer !== undefined) clearInterval(this.timer);
    this.timer = undefined;
    this.transport.stop();
  }

  /** The chats bound, by id. */
  bindings(): string[] {
    const dir = join(this.home, "phone");
    return existsSync(dir) ? readdirSync(dir).filter((n) => n.endsWith(".json")).map((n) => n.slice(0, -".json".length)).sort() : [];
  }

  private path(chat: string): string { return join(this.home, "phone", `${chat.replace(/[^A-Za-z0-9_-]/g, "_")}.json`); }
  private read(chat: string): Binding | undefined {
    const p = this.path(chat);
    if (!existsSync(p)) return undefined;
    try { return JSON.parse(readFileSync(p, "utf8")) as Binding; } catch { return undefined; }
  }
  private write(chat: string, b: Binding): void { writeAtomic(this.path(chat), `${JSON.stringify(b)}\n`); }
  private client(token?: string): RelayClient { return new RelayClient(this.relayUrl, token, this.fetchImpl); }

  /** A text from the phone: /bind with the code, /tell to an agent, a reply to a card (自己写), or the one help line. */
  async onText(chat: string, text: string, replyTo?: string): Promise<void> {
    const bind = /^\/(?:start|bind)\s+(\d{6})\s*$/.exec(text.trim());
    if (bind !== null) {
      let r: { status: number; body: unknown };
      try { r = await this.client().post("/phone/bind", { code: bind[1] }); } catch (e) { await this.transport.send(chat, `绑定失败：${e instanceof Error ? e.message : String(e)}`); return; }
      if (r.status !== 200) { await this.transport.send(chat, `绑定失败：${reasonOf(r)}`); return; }
      const { token, person } = r.body as { token: string; person: string };
      this.write(chat, { person, token, seen: [], cards: {}, n: 0 });
      await this.transport.send(chat, `已绑定 ${person}\n${FIRST_LINE}`);
      await this.mirror(chat);
      return;
    }
    const b = this.read(chat);
    if (b === undefined) { await this.transport.send(chat, BIND_FIRST); return; }
    const tell = /^\/tell\s+(\S+)\s+([\s\S]+)$/.exec(text.trim());
    if (tell !== null) {
      let r: { status: number; body: unknown };
      try { r = await this.client(b.token).post("/tell", { agent: tell[1], text: tell[2]!.trim() }); } catch (e) { await this.transport.send(chat, `没进队列：${e instanceof Error ? e.message : String(e)}`); return; }
      await this.transport.send(chat, r.status === 201 ? tellReply(tell[1]!, b.person) : `没进队列：${reasonOf(r)}`);
      return;
    }
    if (replyTo !== undefined) {
      const card = Object.values(b.cards).find((c) => c.message === replyTo);
      if (card !== undefined) { await this.answer(chat, b, card.key, "自己写", text.trim()); return; }
    }
    await this.transport.send(chat, HELP_LINE);
  }

  /** A button pressed: the card's action, by the data `<card>:<i>`. A press on nothing says nothing. */
  async onPress(chat: string, data: string): Promise<void> {
    const m = /^(\d+):(\d+)$/.exec(data);
    const b = this.read(chat);
    if (m === null || b === undefined) return;
    const card = b.cards[m[1]!];
    const action = card?.actions[Number(m[2])];
    if (card === undefined || action === undefined) return;
    await this.answer(chat, b, card.key, action);
  }

  /** The answer through the relay as the person (transport §7.2 回流为 human 来源的证据), and the bot's word on it. */
  private async answer(chat: string, b: Binding, key: string, action: string, text?: string): Promise<void> {
    let r: { status: number; body: unknown };
    try { r = await this.client(b.token).post("/inbox/answer", { key, action, ...(text !== undefined ? { text } : {}) }); } catch (e) { await this.transport.send(chat, `没答上：${e instanceof Error ? e.message : String(e)}`); return; }
    if (r.status !== 200) { await this.transport.send(chat, `没答上：${reasonOf(r)}`); return; }
    const said = (r.body as { said?: string }).said;
    await this.transport.send(chat, `已答：${action}${said !== undefined && said !== action ? `\n${said}` : ""}`);
  }

  /** The inbox mirrored (transport §7.2 镜像): every need not shown yet becomes a card with its buttons; how many went out. */
  async mirror(only?: string): Promise<number> {
    let n = 0;
    for (const chat of this.bindings()) {
      if (only !== undefined && chat !== only) continue;
      const b = this.read(chat);
      if (b === undefined) continue;
      let needs: Need[];
      try { const r = await this.client(b.token).get("/inbox"); if (r.status !== 200) continue; needs = r.body as Need[]; } catch { continue; }
      for (const need of needs) {
        if (b.seen.includes(need.key)) continue;
        b.n++;
        const id = String(b.n);
        const buttons = need.actions.map((label, i) => ({ label, data: `${id}:${i}` }));
        const message = await this.transport.send(chat, `[${NEED_VERBS[need.kind]}] ${need.text}`, buttons.length > 0 ? [buttons] : undefined);
        b.cards[id] = { key: need.key, actions: need.actions, message };
        b.seen.push(need.key);
        n++;
      }
      if (b.seen.length > SEEN_LIMIT) b.seen = b.seen.slice(-SEEN_LIMIT);
      this.write(chat, b);
    }
    return n;
  }
}
```

```ts
// Telegram as the bridge's transport (transport §7.2 先接 Telegram): long
// polling on getUpdates, a text with the message it replies to, a button as
// a callback query answered at once, a message with an inline keyboard.
// Only fetch; a fake one drives the tests.
import type { Button, Transport } from "./bridge.js";

interface Update { update_id: number; message?: { message_id: number; chat: { id: number | string }; text?: string; reply_to_message?: { message_id: number } }; callback_query?: { id: string; from: { id: number | string }; message?: { chat: { id: number | string }; message_id: number }; data?: string } }
type Handlers = Parameters<Transport["start"]>[0];

export class TelegramTransport implements Transport {
  private offset = 0;
  private stopped = true;
  private handlers: Handlers | undefined;
  private loop: Promise<void> = Promise.resolve();

  constructor(private readonly botToken: string, private readonly fetchImpl: typeof fetch = fetch, private readonly retryMs = 3000) {}

  private async call(method: string, body: Record<string, unknown>): Promise<unknown> {
    const r = await this.fetchImpl(`https://api.telegram.org/bot${this.botToken}/${method}`, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) });
    const j = (await r.json()) as { ok: boolean; result?: unknown; description?: string };
    if (!j.ok) throw new Error(j.description ?? `telegram ${method} failed`);
    return j.result;
  }

  start(h: Handlers): void {
    this.handlers = h;
    this.stopped = false;
    this.loop = this.poll();
  }

  stop(): void { this.stopped = true; }

  /** Once the polling loop has ended (tests wait on it). */
  idle(): Promise<void> { return this.loop; }

  private async poll(): Promise<void> {
    while (!this.stopped) {
      try {
        const updates = (await this.call("getUpdates", { offset: this.offset, timeout: 25 })) as Update[];
        for (const u of updates) { this.offset = u.update_id + 1; this.dispatch(u); }
      } catch {
        await new Promise((done) => setTimeout(done, this.retryMs));
      }
    }
  }

  private dispatch(u: Update): void {
    const h = this.handlers;
    if (h === undefined) return;
    if (u.message?.text !== undefined) void h.onText(String(u.message.chat.id), u.message.text, u.message.reply_to_message !== undefined ? String(u.message.reply_to_message.message_id) : undefined);
    if (u.callback_query !== undefined) {
      void h.onPress(String(u.callback_query.message?.chat.id ?? u.callback_query.from.id), u.callback_query.data ?? "");
      void this.call("answerCallbackQuery", { callback_query_id: u.callback_query.id }).catch(() => undefined);
    }
  }

  async send(chat: string, text: string, buttons?: Button[][]): Promise<string> {
    const r = (await this.call("sendMessage", { chat_id: chat, text, ...(buttons !== undefined ? { reply_markup: { inline_keyboard: buttons.map((row) => row.map((b) => ({ text: b.label, callback_data: b.data }))) } } : {}) })) as { message_id: number };
    return String(r.message_id);
  }
}
```
