# IM Plan K: Split Cards, the One-Line Brief, the Declaration of 接单

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** The rest of ④: 拆分 (§8.4: the worker's or the delegator's agent splits an in-progress card into child cards, the parent stays in progress, the children's budgets sum to no more than the parent's, two levels at most, the whole tree visible to the delegator); the 委托书 drafted from one line in the DM and 改一下 before 发布 (§8.1); the 接单声明 on the agent's page (§9.2: 资格 as the host reports it, 时段, 预算, 接什么) with 接单模式 the owner opens, the claim refused while it is closed or outside the hours, and the card saying 等它有资格的会话 · 每天 22:00 到 8:00; a hosted agent 天然 eligible.

**Architecture:** `card.ts` grows the verbs `edit`, `split`, `tree`, a card's `children`, and the fold of `edit-N` and `split-N` blocks; `src/core/im/declare.ts` (pure) holds the declaration, its hours (with a time zone), the four lines and the eligibility words. The relay edits a draft, splits a card into drafts under the splitter's owner, folds a tree, keeps `declare-N` on the agent's page, and refuses a claim as the declaration says. The brief is one op, `im-brief`: the bridge makes the owner's DM with the agent when it is not there and says the line into it as the owner — the agent's turn then drafts with `im_card new`, and 改一下 under a draft says the words the same way; the board gets the input under the cards. The CLI gets `card edit|split|tree`, `declare`, `profile`.

**Tech Stack:** TypeScript (`src/core` pure, `src/relay`, `src/daemon`, `src/bridge.ts`, `src/hosts/claude-code` pure, `hooks/register.tsx`), `node:test` over `dist/`, `claude plugin test` for the hooks.

**Before this plan:** plans A–J are on `main` (0.3.3).

// The declaration of 接单 on an agent's page (§9.2): whether the owner
// opened 接单模式, the hours (a wall-clock span in a zone, over midnight if
// it must), the day's token budget, what it takes; and the eligibility the
// host reports — an API or self-hosted session may claim, a subscription one
// never, a hosted agent always. Pure: the record, the rule, the words.

export interface Hours { from: string; to: string; tz: string }
export interface Declaration { open: boolean; hours?: Hours; budget?: number; kinds: string[]; skills: string[]; network: boolean }
export const DEFAULT_DECLARATION: Declaration = { open: false, kinds: [], skills: [], network: false };

const obj = (v: unknown): Record<string, unknown> | undefined => (typeof v === "object" && v !== null && !Array.isArray(v) ? (v as Record<string, unknown>) : undefined);
const strings = (v: unknown): string[] | undefined => (Array.isArray(v) ? v.filter((x): x is string => typeof x === "string" && x.trim() !== "").map((x) => x.trim()) : undefined);
const HOURS = /^(\d{1,2}):(\d{2})-(\d{1,2}):(\d{2})(Z|[+-]\d{2}:\d{2})$/;
const pad = (n: string): string => n.padStart(2, "0");

/** 'HH:MM-HH:MM' followed by a zone (`Z` or `±HH:MM`); undefined for anything else. */
export function hoursOf(v: unknown): Hours | undefined {
  const m = typeof v === "string" ? HOURS.exec(v.trim()) : null;
  if (m === null) return undefined;
  const [, fh, fm, th, tm, tz] = m;
  if (Number(fh) > 23 || Number(th) > 23 || Number(fm) > 59 || Number(tm) > 59) return undefined;
  return { from: `${pad(fh!)}:${fm}`, to: `${pad(th!)}:${tm}`, tz: tz! };
}

/** A declaration from a body, each field given replacing the current's, a bad one falling to the default; `hours: null` clears the hours. */
export function declarationOf(v: unknown, current: Declaration = DEFAULT_DECLARATION): Declaration {
  const o = obj(v) ?? {};
  const out: Declaration = { ...current, kinds: [...current.kinds], skills: [...current.skills] };
  if (typeof o["open"] === "boolean") out.open = o["open"];
  if (o["hours"] === null) delete out.hours;
  else if (o["hours"] !== undefined) { const h = hoursOf(o["hours"]); if (h !== undefined) out.hours = h; else delete out.hours; }
  if (o["budget"] === null) delete out.budget;
  else if (o["budget"] !== undefined) { const b = o["budget"]; if (typeof b === "number" && Number.isInteger(b) && b >= 0) out.budget = b; else delete out.budget; }
  const kinds = strings(o["kinds"]); if (o["kinds"] !== undefined) out.kinds = kinds ?? [];
  const skills = strings(o["skills"]); if (o["skills"] !== undefined) out.skills = skills ?? [];
  if (typeof o["network"] === "boolean") out.network = o["network"];
  return out;
}

const minutesOf = (hhmm: string): number => Number(hhmm.slice(0, 2)) * 60 + Number(hhmm.slice(3, 5));
const offsetOf = (tz: string): number => (tz === "Z" ? 0 : (tz.startsWith("-") ? -1 : 1) * (Number(tz.slice(1, 3)) * 60 + Number(tz.slice(4, 6))));

/** Whether the instant falls in the hours by the zone's wall clock; the start counts, the end does not; a span over midnight wraps. No hours: always. */
export function withinHours(now: string, hours: Hours | undefined): boolean {
  if (hours === undefined) return true;
  const t = Date.parse(now);
  if (Number.isNaN(t)) return false;
  const wall = (((t / 60_000 + offsetOf(hours.tz)) % 1440) + 1440) % 1440;
  const from = minutesOf(hours.from), to = minutesOf(hours.to);
  return from <= to ? wall >= from && wall < to : wall >= from || wall < to;
}

export const hoursWords = (h: Hours | undefined): string => (h === undefined ? "随时" : `每天 ${h.from} 到 ${h.to}`);
/** The hours as a body writes them, for a record that `declarationOf` reads back. */
export const hoursString = (h: Hours): string => `${h.from}-${h.to}${h.tz}`;

/** What the host's sessions say (§9.2): a hosted agent is always eligible; else the best login among its sessions. */
export function eligibilityWords(logins: readonly string[], hosted: boolean): string {
  if (hosted) return "托管 · 天然有资格";
  if (logins.includes("api")) return "API 会话 · 有资格";
  if (logins.includes("self-hosted")) return "自托管会话 · 有资格";
  if (logins.length > 0) return "只有订阅登录的会话 · 不显示接单入口";
  return "还没有会话";
}

/** The four lines on the agent's page (§9.2), and a fifth when 接单模式 is off. */
export function declarationLines(d: Declaration, eligibility: string): string[] {
  const what = [...(d.kinds.length > 0 ? [d.kinds.join("、")] : []), ...(d.skills.length > 0 ? [`skill ${d.skills.join("、")}`] : [])];
  return [
    `资格：${eligibility}`,
    `时段：${hoursWords(d.hours)}${d.hours !== undefined ? " · 主人睡了之后" : ""}`,
    `预算：${d.budget === undefined ? "不限" : `每天 ${d.budget} token · 推理走平台池`}`,
    `接什么：${what.length > 0 ? what.join(" · ") : "没说"} · ${d.network ? "可联网" : "不联网"}`,
    ...(d.open ? [] : ["接单模式：关"]),
  ];
}

/** What a card published to a named agent waits for (§9.2): the owner's 接单模式, or an eligible session in the hours. */
export const waitingForSession = (d: Declaration): string => (d.open ? `等它有资格的会话${d.hours === undefined ? "" : ` · ${hoursWords(d.hours)}`}` : "等主人开接单模式");
```

  private cardAt(address: string): { owner: string; card: Card } | undefined {
    const a = parseAddress(address);
    const m = a === undefined || a.relay !== this.name ? null : /^cards\/([^/]+)\.geml$/.exec(a.doc);
    const card = a === undefined || m === null ? undefined : this.taskOf(a.owner, m[1]!);
    return card === undefined ? undefined : { owner: a!.owner, card };
  }

  /** 改一下 (§8.1): the delegator or their agent changes a draft's fields — an edit block the fold lays over the head. */
  editTask(by: TokenRecord, owner: string, id: string, body: unknown): Result<{ card: Card }> {
    const card = this.taskOf(owner, id);
    if (card === undefined) return fail(404, "没有这张卡");
    if (!rolesOf(card, this.actorOf(by)).has("delegator")) return fail(403, "只有委托人或其 agent 能改委托书");
    if (card.state !== "draft") return fail(409, `卡在${STATE_WORDS_OF(card.state)}，发布后不能改；撤回再改`);
    const fields = Object.fromEntries(Object.entries(obj(body) ?? {}).filter(([k]) => EDITABLE.has(k)));
    if (Object.keys(fields).length === 0) return fail(400, "没有可改的字段");
    const faults = cardFaults({ ...card, ...fields });
    if (faults.length > 0) return fail(400, faults.join("；"));
    const doc = `cards/${id}.geml`;
    const a = appendBlock(this.root, owner, doc, nextId("edit"), (_d, prev, bid) => sealed({ id: bid, ...(fields as Record<string, JsonValue>), by: by.speaker, at: this.now() }, prev));
    this.emit(owner, doc, a.block);
    return { ok: true, card: this.taskOf(owner, id)! };
  }

  /** How many parents a card has above it (§8.4: two levels at most). */
  private depthOf(card: Card): number {
    let n = 0, up = card.parent;
    while (up !== undefined && n < 8) {
      const parent = this.cardAt(up);
      if (parent === undefined) break;
      n++; up = parent.card.parent;
    }
    return n;
  }

  /** 拆分 (§8.4): the worker (or its owner) or the delegator's agent splits an in-progress card into drafts under the splitter's own name, each with the parent; the children so far and the new ones sum to no more than the parent's budget; two levels at most; the parent stays 进行中 and lists them. */
  splitTask(by: TokenRecord, owner: string, id: string, body: unknown): Result<{ card: Card; children: { owner: string; id: string; address: string }[] }> {
    const card = this.taskOf(owner, id);
    if (card === undefined) return fail(404, "没有这张卡");
    if (card.state !== "in-progress") return fail(409, `卡在${STATE_WORDS_OF(card.state)}，只有进行中的卡能拆`);
    const roles = rolesOf(card, this.actorOf(by));
    if (!roles.has("worker") && !roles.has("delegator")) return fail(403, "只有接单方或委托人的 agent 能拆卡");
    const splitter = this.actorOf(by).person;
    if (splitter === undefined) return fail(403, "拆卡的要有主人");
    const raw = Array.isArray(obj(body)?.["children"]) ? (obj(body)!["children"] as unknown[]) : [];
    if (raw.length === 0) return fail(400, "子卡至少一张");
    const parentAddress = this.cardAddress(owner, id);
    const inputs = raw.map((c) => ({ ...(obj(c) ?? {}), parent: parentAddress }));
    const faults = inputs.flatMap((c, i) => { const f = cardFaults(c); return f.length > 0 ? [`第 ${i + 1} 张子卡：${f.join("；")}`] : []; });
    if (faults.length > 0) return fail(400, faults.join("；"));
    const children = inputs.map((c) => cardInput(c)!);
    const sofar = card.children.map((addr) => this.cardAt(addr)?.card).filter((c): c is Card => c !== undefined);
    const fault = splitFault(card, [...sofar, ...children], this.depthOf(card));
    if (fault !== undefined) return fail(409, fault);
    const made: { owner: string; id: string; address: string }[] = [];
    for (const input of children) {
      for (let tries = 0; tries < 16; tries++) {
        const cid = `t-${randomBytes(2).toString("hex")}`;
        const doc = `cards/${cid}.geml`;
        if (readDoc(this.root, splitter, doc).exists) continue;
        const a = appendSealed(this.root, splitter, doc, "card", { kind: "task", ...input, by: `person:${splitter}`, at: this.now() } as unknown as Record<string, JsonValue>);
        this.emit(splitter, doc, a.block);
        made.push({ owner: splitter, id: cid, address: this.cardAddress(splitter, cid) });
        break;
      }
    }
    if (made.length !== children.length) return fail(500, "卡号都被占了");

    const a = appendBlock(this.root, owner, doc, nextId("split"), (_d, prev, bid) => sealed({ id: bid, by: by.speaker, children: made.map((m) => m.address), at: this.now() }, prev));
    this.emit(owner, doc, a.block);
    return { ok: true, card: this.taskOf(owner, id)!, children: made };
  }

[] : c.children.map((addr) => this.cardAt(addr)).filter((x): x is { owner: string; card: Card } => x !== undefined).map((x) => fold(x.owner, x.card, depth + 1)) });

```ts
// 一句话起草委托书 (§8.1): the owner's line goes into their DM with the
// agent — made on the relay when there is none yet — as the owner's own
// word, so the agent's next turn drafts the card with im_card new; 改一下
// under a draft says its words the same way. One look at the relay, then
// the outbox carries the word like any other.
import { RelayClient, RelayDown, reasonOf } from "./client.js";
import { say } from "./deliver.js";
import { readConfig } from "./home.js";

export const briefText = (text: string): string => `起草一张委托书：${text}。用 im_card new 起草成草稿给我过目；我说「改一下」就用 im_card edit 改，我点发布才发。`;
export const changeText = (id: string, words: string): string => `改一下 #${id}：${words}。用 im_card edit 改草稿，改完说一声。`;

export async function briefCall(home: string, o: { text: string; agent?: string; card?: string }, now: string): Promise<{ ok: true; venue: string } | { ok: false; reason: string }> {
  const c = readConfig(home);
  if (c === undefined) return { ok: false, reason: "没有 connect.json：先 gemworks connect setup" };
  const agent = o.agent ?? Object.keys(c.agents)[0];
  if (agent === undefined) return { ok: false, reason: "本会话没有 agent：在设置里选一个" };
  if (c.agents[agent] === undefined) return { ok: false, reason: `没有 ${agent} 的令牌` };
  const id = `dm-${agent}`;
  const client = new RelayClient(c.relay.url, c.token);
  try {
    const have = await client.get(`/${encodeURIComponent(c.person)}/venues/${id}.geml`);
    if (have.status === 404) {
      const made = await client.post("/venues", { id, kind: "dm", name: agent, agents: [`agent:${agent}@${c.person}`] });
      if (made.status !== 201) return { ok: false, reason: reasonOf(made) };
    } else if (have.status !== 200) return { ok: false, reason: reasonOf(have) };
  } catch (e) { return { ok: false, reason: e instanceof RelayDown ? `连不上中继：${e.message}` : e instanceof Error ? e.message : String(e) }; }
  say(home, { as: "person", owner: c.person, venue: id, text: o.card === undefined ? briefText(o.text) : changeText(o.card, o.text) }, now);
  return { ok: true, venue: `${c.person}/${id}` };
}
